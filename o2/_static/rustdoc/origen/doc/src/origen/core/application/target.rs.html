1//! A default target/env for an application can be set by config/application.toml, this can
2//! then be overridden by the user by setting a temporary workspace default via the "origen t"
3//! and "origen e" commands.
4//! These commands store the user's selections in .origen/application.toml and this file should
5//! NOT be checked into revision control.
6//!
7//! The target can be further overridden for a particular origen command invocation via
8//! the -t and -e options, or programmatically within the application code, however that is all
9//! handled on the front end in Python code.

21#[macro_export]
22macro_rules! clean_target {
23    ($name:expr, $dir:expr, $return_file:expr) => {{
24        $crate::core::application::target::clean_name(
25            $name,
26            $dir,
27            $return_file,
28            &$crate::app().unwrap().root,
29        )
30    }};
31}
32
33macro_rules! toml {
34    ($root:expr) => {{
35        $root.join(".origen").join("application.toml")
36    }};
37}

39/// Sanitizes the given target/env name and returns it, but will exit the process if it does
40/// not uniquely identify a single target/env file.
41/// Set the last arg to true to return the path to the matching target instead.
42pub fn clean_name(name: &str, dir: &str, return_file: bool, root: &PathBuf) -> String {
43    let matches = matches(name, dir, root);
44    let t = dir.trim_end_matches("s");
45
46    if matches.len() == 0 {
47        println!(
48            "No matching {} '{}' found, here are the available {}s:",
49            t, name, t
50        );
51        for file in all(&root.join(dir)).iter() {
52            println!(
53                "    {}",
54                diff_paths(&file, &root.join(dir)).unwrap().display()
55            );
56        }
57    } else if matches.len() > 1 {
58        println!(
59            "That {} name '{}' is ambiguous, please try again to narrow it down to one of these:",
60            t, name
61        );
62        for file in matches.iter() {
63            println!(
64                "    {}",
65                diff_paths(&file, &root.join(dir)).unwrap().display()
66            );
67        }
68    } else {
69        if return_file {
70            return format!("{}", matches[0].display());
71        } else {
72            let clean = format!(
73                "{}",
74                diff_paths(&matches[0], &root.join(dir)).unwrap().display()
75            );
76            return clean;
77        }
78    }
79    exit(1);
80}

82/// Returns an array of possible target/environment files that match the given name/snippet
83// TODO: look into updating this to use the PathBuf PartialEq Trait to compare instead string compare which is prone to bugs due to OS differences
84pub fn matches(name: &str, dir: &str, root: &PathBuf) -> Vec<PathBuf> {
85    let mut files: Vec<PathBuf> = Vec::new();
86
87    for file in WalkDir::new(format!("{}", root.join(dir).display())) {
88        let path = file.unwrap().into_path();
89        if path.is_file() {
90            let mut path_str = format!("{}", diff_paths(&path, &root.join(dir)).unwrap().display());
91            // in case we're running on Windows normalize to linux style path separator character
92            path_str = path_str.replace("\\", "/").replace("//", "/");
93
94            if path_str.contains(name) {
95                files.push(path);
96            // Try again without the leading dir in case the user has supplied a path
97            } else {
98                let re = Regex::new(format!(r#".*{}(\\|/)"#, dir).as_str()).unwrap();
99                let mut new_name: String = re.replace_all(&name, "").into();
100
101                // in case we're running on Windows normalize to linux style path separator character
102                new_name = new_name.replace("\\", "/").replace("//", "/");
103
104                if path_str.contains(&new_name) {
105                    files.push(path);
106                }
107            }
108        }
109    }
110    // After collecting all the matches, if the size > 1 then filter again for exact matches
111    if files.len() > 1 {
112        files = files
113            .into_iter()
114            .filter(|path| path.file_name().unwrap().to_str().unwrap() == &format!("{}.py", name))
115            .collect();
116    }
117    files
118}

120/// Gets the currently enabled targets
121pub fn get(full_paths: bool) -> Option<Vec<String>> {
122    app()
123        .unwrap()
124        .with_config_mut(|config| {
125            config.refresh();
126            match config.target.as_ref() {
127                Some(targets) => Ok(Some(
128                    targets
129                        .iter()
130                        .map(|t| clean_target!(t, "targets", full_paths))
131                        .collect::<Vec<String>>()
132                        .clone(),
133                )),
134                None => Ok(None),
135            }
136        })
137        .unwrap()
138}

145pub fn set_at_root(targets: Vec<&str>, root: &PathBuf) -> Vec<String> {
146    let mut to_set = vec![];
147    for t in targets.iter() {
148        let cn = clean_name(t, "targets", true, root);
149        if to_set.contains(&cn) {
150            log_error!(
151                "Target '{}' appears multiple times in the TARGETS list ({})",
152                t,
153                cn
154            );
155            exit(1);
156        }
157        to_set.push(cn.clone());
158    }
159    set_workspace_array_at_root("target", to_set.clone(), root);
160    to_set
161}

173/// Enables additional targets in the workspace
174pub fn add(targets: Vec<&str>) {
175    let mut current: Vec<String> = app()
176        .unwrap()
177        .with_config(|config| {
178            let c = match &config.target {
179                Some(targets) => targets.clone(),
180                None => vec![],
181            }
182            .iter()
183            .map(|t| clean_target!(t, "targets", true))
184            .collect();
185            Ok(c)
186        })
187        .unwrap();
188
189    let mut added = Vec::new();
190    for t in targets.iter() {
191        // Check that the targets to add are valid
192        let clean_t = clean_target!(t, "targets", true);
193
194        // If the target is already added, remove it from its current position and reapply it in the order
195        // given here
196        current.retain(|c| *c != clean_t);
197
198        if added.contains(&clean_t) {
199            log_error!(
200                "Target '{}' appears multiple times in the TARGETS list ({})",
201                t,
202                clean_t
203            );
204            exit(1);
205        }
206        added.push(clean_t);
207    }
208    current.extend(added);
209
210    set_workspace_array("target", current);
211}

213/// Disables currently enables targets in the workspace
214pub fn remove(targets: Vec<&str>) {
215    let mut current: Vec<String> = app()
216        .unwrap()
217        .with_config(|config| {
218            let c: Vec<String> = match &config.target {
219                Some(targets) => targets.clone(),
220                None => vec![],
221            }
222            .iter()
223            .map(|t| clean_target!(t, "targets", true))
224            .collect();
225            Ok(c)
226        })
227        .unwrap();
228
229    let mut removed = HashSet::new();
230    let mut len = current.len();
231    for t in targets.iter() {
232        let clean_t = clean_target!(t, "targets", true);
233
234        if removed.contains(&clean_t) {
235            log_error!(
236                "Target '{}' appears multiple times in the TARGETS list ({})",
237                t,
238                clean_t
239            );
240            exit(1);
241        }
242
243        current.retain(|c| *c != clean_t);
244        let new_len = current.len();
245        if new_len == len {
246            log_error!("Tried to remove non-activated target '{}' ({})", t, clean_t);
247            exit(1);
248        }
249        len = new_len;
250        removed.insert(clean_t);
251    }
252
253    if current.len() == 0 {
254        println!("All targets were removed. Resetting to the default target.");
255        reset();
256    } else {
257        set_workspace_array("target", current);
258    }
259}

261/// Returns all files from the given directory
262pub fn all(dir: &PathBuf) -> Vec<PathBuf> {
263    let mut files: Vec<PathBuf> = Vec::new();
264
265    for file in WalkDir::new(dir).sort_by_file_name() {
266        let path = file.unwrap().into_path();
267        if path.is_file() && path.extension().unwrap_or("".as_ref()) == "py" {
268            files.push(path);
269        }
270    }
271    files
272}

274/// Sets the given key and value (currently only a string is supported) in
275/// .origen/application.toml
276pub fn set_workspace(key: &str, val: &str) {
277    ensure_app_dot_toml();
278    delete_val(key);
279    add_val(key, val);
280}

293/// Deletes the given key (and its val) from .origen/application.toml if it exists
294pub fn delete_val(key: &str) {
295    delete_val_at_root(key, &app().unwrap().root);
296}
297
298pub fn delete_val_at_root(key: &str, root: &PathBuf) {
299    let path = toml!(root);
300    let data =
301        fs::read_to_string(&path).expect(&format!("Unable to read file {}", &path.display()));
302    let re = Regex::new(format!(r#"{}\s?=.*(\r\n|\n)?"#, escape(key)).as_str()).unwrap();
303    let new_data: String = re.replace_all(&data, "").into();
304    fs::write(&path, new_data).expect(&format!("Unable to write file {}", &path.display()));
305}

307/// Appends the given key/val pair to the end of .origen/application.toml
308fn add_val(key: &str, val: &str) {
309    let path = app().unwrap().root.join(".origen").join("application.toml");
310    let data = fs::read_to_string(path).expect("Unable to read file .origen/application.toml");
311    let new_data = format!("{}\n{} = \"{}\"", data.trim(), key, val);
312    fs::write(
313        app().unwrap().root.join(".origen").join("application.toml"),
314        new_data,
315    )
316    .expect("Unable to write file .origen/application.toml!");
317}

319/// Appends the given key/val pair to the end of .origen/application.toml

324fn add_val_array_at_root(key: &str, vals: Vec<String>, root: &PathBuf) {
325    let path = toml!(root);
326    let data =
327        fs::read_to_string(&path).expect(&format!("Unable to read file {}", &path.display()));
328
329    // Note: use string literals here to account for Windows paths
330    let new_data = format!(
331        "{}\n{} = [{}]",
332        data.trim(),
333        key,
334        vals.iter()
335            .map(|v| format!("'{}'", v))
336            .collect::<Vec<String>>()
337            .join(", ")
338    );
339    fs::write(&path, new_data).expect(&format!("Unable to write file {}", &path.display()));
340}

342/// Verifies that .origen/application.toml exists and if not creates one
343fn ensure_app_dot_toml() {
344    ensure_app_dot_toml_at_root(&app().unwrap().root)
345}
346
347fn ensure_app_dot_toml_at_root(root: &PathBuf) {
348    let path = root.join(".origen");
349    if !path.exists() {
350        fs::create_dir(&path).expect("Unable to create directory .origen!");
351    }
352    let path = path.join("application.toml");
353    if !path.exists() {
354        let data =
355            "# This file is generated by Origen and should not be checked into revision control";
356        fs::write(&path, data).expect(&format!("Unable to write file {}", &path.display()));
357    }
