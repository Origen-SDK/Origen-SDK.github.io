1//! The file_handler is responsible for processing the file arguments supplied
2//! to Origen commands from the CLI.
3//! It provides methods for consumers to retreive one file at a time or all files
4//! at once and seamlessly opens up lists to get to the inidividual files inside.
5

17lazy_static! {
18    static ref FILES: Mutex<Files> = Mutex::new(Files::new());

21#[derive(Debug)]
22pub struct File {
23    pub file_obj: StdFile,
24    pub path: PathBuf,
25}
26
27impl std::clone::Clone for File {
28    fn clone(&self) -> Self {
29        Self::open(self.path.clone()).unwrap()
30    }
31}
32
33impl File {
34    /// Creates or overrides the given file and panics if the creation fails
35    pub fn create(f: PathBuf) -> Self {
36        Self::mkdir_p(
37            &f.parent()
38                .expect(&format!("Unable to locate parent of {:?}", f)),
39        );
40        let file = StdFile::create(f.clone()).expect(&format!("Unable to create file at {:?}", f));
41        Self {
42            file_obj: file,
43            path: f.clone(),
44        }
45    }
46
47    /// Attempts to open the given file for reading, returning an error if the open fails
48    pub fn open(f: PathBuf) -> Result<Self> {
49        let file = StdFile::open(f.clone()).expect(&format!("Unable to open file at {:?}", f));
50        Ok(Self {
51            file_obj: file,
52            path: f,
53        })
54    }
55
56    pub fn mkdir_p(path: &Path) {
57        create_dir_all(path).expect(&format!("Unable to create all directories at {:?}", path));
58    }
59
60    /// Writes content to the underlying file and panics if the write failed
61    pub fn write(&mut self, content: &str) {
62        self.file_obj
63            .write_all(content.as_bytes())
64            .expect(&format!("Error writing file {:?}", self.path));
65    }
66
67    pub fn write_ln(&mut self, content: &str) {
68        self.file_obj
69            .write_all(format!("{}\n", content).as_bytes())
70            .expect(&format!("Error writing file {:?}", self.path));
71    }
72}

74#[derive(Debug)]
75/// This struct is used as a singleton to store a permanent record of the file arguments
76/// given to the current command and the clean list of files this resolves to
77struct Files {
78    /// The file arguments as originally supplied
79    items: Vec<String>,
80    /// The resultant list of files from resolving any lists in the original args
81    files: Vec<PathBuf>,
82}
83
84impl Files {
85    fn new() -> Files {
86        Files {
87            items: Vec::new(),
88            files: Vec::new(),
89        }
90    }
91
92    /// Load a new set of file arguments
93    fn init(&mut self, mut files: Vec<String>) -> Result<()> {
94        // Convert any / paths to \
95        if cfg!(target_os = "windows") {
96            files = files
97                .iter()
98                .map(|f| format!("{}", PathBuf::from_slash(f).display()))
99                .collect();
100        }
101        self.items.clear();
102        self.items.append(&mut files);
103        self.files.clear();
104        self.resolve()?;
105        Ok(())
106    }
107
108    fn file(&mut self, index: usize) -> Option<PathBuf> {
109        if index < self.files.len() {
110            Some(self.files[index].clone())
111        } else {
112            None
113        }
114    }
115
116    // Eventually this will open up list and directory args
117    fn resolve(&mut self) -> Result<()> {
118        let items = self.items.clone();
119        for item in &items {
120            match Path::new(item).canonicalize() {
121                Ok(x) => {
122                    if x.is_dir() {
123                        self.resolve_dir(&x)?;
124                    } else {
125                        self.files.push(x);
126                    }
127                }
128                Err(err) => bail!("{} - {}", err.to_string(), item),
129            }
130        }
131        Ok(())
132    }
133
134    fn resolve_dir(&mut self, dir: &Path) -> Result<()> {
135        for entry in fs::read_dir(dir)? {
136            let entry = entry?;
137            let path = entry.path();
138            if path.is_dir() {
139                self.resolve_dir(&path)?;
140            } else {
141                self.files.push(path);
142            }
143        }
144        Ok(())
145    }
146}

148#[derive(Debug, Clone)]
149/// This is an iterator + external API for consuming the file list
150pub struct FileHandler {
151    i: usize,
152}
153
154impl FileHandler {
155    pub fn new() -> FileHandler {
156        FileHandler { i: 0 }
157    }
158
159    pub fn init(&mut self, files: Vec<String>) -> Result<()> {
160        let mut f = FILES.lock().unwrap();
161        f.init(files)?;
162        self.i = 0;
163        Ok(())
164    }
165
166    pub fn len(&self) -> usize {
167        let f = FILES.lock().unwrap();
168        f.files.len()
169    }
170}
171
172impl Iterator for FileHandler {
173    type Item = PathBuf;
174
175    fn next(&mut self) -> Option<PathBuf> {
176        let mut f = FILES.lock().unwrap();
177        let r = f.file(self.i);
178        self.i += 1;
179        r
180    }
181}
