20/// The DUT stores all objects associated with a particular device.
21/// Each object type is organized into vectors, where a particular object's position within the
22/// vector is considered its unique ID.
23/// A register then (for example) does not embed its bit objects but rather contains a list of
24/// bit IDs. This approach allows bits to be easily passed around by ID to enable the creation of
25/// bit collections that are small (a subset of a register's bits) or very large (all bits in
26/// a memory map).
27#[derive(Debug, IdGetters)]
28#[id_getters_by_mapping(
29    field = "timeset",
30    parent_field = "models",
31    return_type = "Timeset",
32    field_container_name = "timesets"
33)]
34#[id_getters_by_mapping(
35    field = "wavetable",
36    parent_field = "timesets",
37    return_type = "Wavetable",
38    field_container_name = "wavetables"
39)]
40#[id_getters_by_mapping(
41    field = "wave_group",
42    parent_field = "wavetables",
43    return_type = "WaveGroup",
44    field_container_name = "wave_groups"
45)]
46#[id_getters_by_mapping(
47    field = "wave",
48    parent_field = "wave_groups",
49    return_type = "Wave",
50    field_container_name = "waves"
51)]
52#[id_getters_by_index(
53    field = "event",
54    parent_field = "waves",
55    return_type = "Event",
56    field_container_name = "wave_events"
57)]
58#[id_getters_by_mapping(
59    field = "pin",
60    parent_field = "models",
61    return_type = "Pin",
62    field_container_name = "pins"
63)]
64#[id_getters_by_mapping(
65    field = "pin_group",
66    parent_field = "models",
67    return_type = "PinGroup",
68    field_container_name = "pin_groups"

133    /// Change the DUT, this replaces the existing mode with a fresh one (i.e.
134    /// deletes all current DUT metadata and state, and updates the name/ID field
135    /// with the given value
136    // This is called once per DUT load
137    pub fn change(&mut self, name: &str) -> Result<()> {
138        crate::with_optional_frontend(|f| f.on_dut_change())?;
139        self.name = name.to_string();
140        self.models.clear();
141        self.memory_maps.clear();
142        self.address_blocks.clear();
143        self.register_files.clear();
144        self.registers.clear();
145        self.accessed_regs.write().unwrap().clear();
146        self.bits.clear();
147        self.timesets.clear();
148        self.wavetables.clear();
149        self.wave_groups.clear();
150        self.waves.clear();
151        self.wave_events.clear();
152        self.id_mappings.clear();
153        self.reg_descriptions.clear();
154        // Add the model for the DUT top-level (always ID 0)
155        let _ = self.create_model(None, "dut", None);
156        Ok(())
157    }

329    /// Create a new model adding it to the existing parent model with the given ID.
330    /// The ID of the newly created model is returned to the caller who should save it
331    /// if they want to access this model directly again (will also be accessible by name
332    /// via the parent model).
333    pub fn create_model(
334        &mut self,
335        parent_id: Option<usize>,
336        name: &str,
337        offset: Option<u128>,
338    ) -> Result<usize> {
339        let id;
340        {
341            id = self.models.len();
342        }
343        {
344            if parent_id.is_some() {
345                let m = self.get_mut_model(parent_id.unwrap())?;
346                if m.sub_blocks.contains_key(name) {
347                    bail!(
348                        "The block '{}' already contains a sub-block called '{}'",
349                        m.display_path(&DUT.lock().unwrap()),
350                        name
351                    );
352                } else {
353                    m.sub_blocks.insert(name.to_string(), id);
354                }
355            }
356        }
357        let new_model = Model::new(id, name.to_string(), parent_id, offset);
358        self.models.push(new_model);
359        Ok(id)
360    }

362    pub fn create_memory_map(
363        &mut self,
364        model_id: usize,
365        name: &str,
366        address_unit_bits: Option<u32>,
367    ) -> Result<usize> {
368        let id;
369        {
370            id = self.memory_maps.len();
371        }
372        {
373            let model = self.get_mut_model(model_id)?;
374
375            if model.memory_maps.contains_key(name) {
376                bail!(
377                    "The block '{}' already contains a memory map called '{}'",
378                    model.name,
379                    name
380                );
381            } else {
382                model.memory_maps.insert(name.to_string(), id);
383            }
384        }
385
386        let mut defaults = MemoryMap::default();
387        match address_unit_bits {
388            Some(v) => defaults.address_unit_bits = v,
389            None => {}
390        }
391        self.memory_maps.push(MemoryMap {
392            id: id,
393            model_id: model_id,
394            name: name.to_string(),
395            ..defaults
396        });
397        Ok(id)
398    }

401        &mut self,
402        memory_map_id: usize,
403        name: &str,
404        offset: Option<u128>,

454    pub fn create_reg(
455        &mut self,
456        address_block_id: usize,
457        register_file_id: Option<usize>,
458        name: &str,
459        offset: usize,
460        size: Option<usize>,
461        bit_order: &str,
462        filename: Option<String>,
463        lineno: Option<usize>,
464        description: Option<String>,
465    ) -> Result<usize> {
466        let id;
467        {
468            id = self.registers.len();
469        }
470        {
471            let a = self.get_mut_address_block(address_block_id)?;
472            if a.registers.contains_key(name) {
473                bail!(
474                    "The address block '{}' already contains a register called '{}'",
475                    a.name,
476                    name
477                );
478            } else {
479                a.registers.insert(name.to_string(), id);
480            }
481        }
482        let mut defaults = Register::default();
483        match size {
484            Some(v) => defaults.size = v,
485            None => {}
486        }
487        match bit_order.parse() {
488            Ok(x) => defaults.bit_order = x,
489            Err(msg) => bail!(&msg),
490        }
491        let reg = Register {
492            id: id,
493            name: name.to_string(),
494            offset: offset,
495            address_block_id: address_block_id,
496            register_file_id: register_file_id,
497            filename: filename,
498            lineno: lineno,
499            description: description,
500            ..defaults
501        };
502
503        self.registers.push(reg);
504        Ok(id)
505    }

530    // Returns the description of this register, if any.
531    // **Note** Adding a description field will override any comment-driven documentation
532    // of a register (ie markdown style comments)
533    pub fn get_reg_description(&mut self, filename: &str, lineno: usize) -> Option<String> {
534        if self.reg_descriptions.get(filename).is_none() {
535            self.parse_descriptions(filename);
536        }
537        match self.reg_descriptions.get(filename) {
538            Some(x) => match x.get(&lineno) {
539                Some(y) => Some(y.to_string()),
540                None => None,
541            },
542            None => None,
543        }
544    }

546    fn parse_descriptions(&mut self, filename: &str) {
547        let path = Path::new(filename);
548        if !path.exists() {
549            return;
550        }
551
552        let f = File::open(path).unwrap();
553        let f = BufReader::new(f);
554        let mut desc = "".to_string();
555
556        let re1 = Regex::new(r"^\s*#\s?(.*)").unwrap();
557        // https://rubular.com/r/QN0aCI8N6Oj77v
558        let re2 = Regex::new(
559            r#"^\s*(SimpleReg|with Reg|with .*\.add_reg|.*\.add_simple_reg|.*\.Field)\(r?f?["'](.*)["']"#,
560        )
561        .unwrap();
562
563        if self.reg_descriptions.get(filename).is_none() {
564            self.reg_descriptions
565                .insert(filename.to_string(), IndexMap::new());
566        }
567        let descs = self.reg_descriptions.get_mut(filename).unwrap();
568
569        let mut i = 1;
570        for line in f.lines() {
571            let line = line.unwrap();
572
573            if re1.is_match(&line) {
574                let caps = re1.captures(&line).unwrap();
575                if desc != "" {
576                    desc += "\n";
577                }
578                desc += caps.get(1).unwrap().as_str();
579            } else if re2.is_match(&line) {
580                if desc != "" {
581                    descs.insert(i, desc);
582                }
583                desc = "".to_string();
584            } else {
585                desc = "".to_string();
586            }
587            i += 1;
588        }
589    }
