1//! A very simple example processor which returns a new version of the
2//! given AST with all comments changed to upper case.
3
4use super::super::nodes::PAT;
5use crate::Result;
6use origen_metal::ast::*;
7
8pub struct UpcaseComments {}
9
10impl UpcaseComments {
11    #[allow(dead_code)]
12    pub fn run(node: &Node<PAT>) -> Result<Node<PAT>> {
13        let mut p = UpcaseComments {};
14        Ok(node.process(&mut p)?.unwrap())
15    }
16}
17
18impl Processor<PAT> for UpcaseComments {
19    fn on_node(&mut self, node: &Node<PAT>) -> origen_metal::Result<Return<PAT>> {
20        match &node.attrs {
21            PAT::Comment(level, msg) => {
22                let new_node = node.replace_attrs(PAT::Comment(*level, msg.to_uppercase()));
23                Ok(Return::Replace(new_node))
24            }
25            _ => Ok(Return::ProcessChildren),
26        }
27    }
28}
29
30#[cfg(test)]
31mod tests {
32    use super::*;
33    use crate::generator::nodes::PAT;
34
35    #[test]
36    fn it_works() {
37        let mut ast = AST::new();
38        ast.push_and_open(node!(PAT::Test, "t1".to_string()));
39        ast.push(node!(PAT::Cycle, 1, false));
40        ast.push(node!(PAT::Comment, 1, "some comment".to_string()));
41
42        let mut expect = AST::new();
43        expect.push_and_open(node!(PAT::Test, "t1".to_string()));
44        expect.push(node!(PAT::Cycle, 1, false));
45        expect.push(node!(PAT::Comment, 1, "SOME COMMENT".to_string()));
46
47        assert_eq!(
48            UpcaseComments::run(&ast.to_node()).expect("Comments upcased"),
49            expect
50        );
51    }
52}
