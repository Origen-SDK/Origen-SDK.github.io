1//! A simple processor which will combine pin actions not separated by a cycle into a single PinAction node
2
3use super::super::nodes::PAT;
4use crate::Result;
5use origen_metal::ast::{Node, Processor, Return};
6use std::collections::HashMap;
7
8/// Combines adjacent pin actions into a single pin action
9pub struct PinActionCombiner {
10    current_state: HashMap<usize, String>,
11    i: usize,
12    first_pass: bool,
13    delete_current_index: bool,
14    indices_to_delete: Vec<usize>,
15}
16
17/// Combines pin actions so that only pin actions which change the pin states are left.
18/// Note: this processor assumes that anything that touches PinAction nodes has already completed.
19/// Because some form of lookahead is needed, and to avoid missing actions that may occur in child nodes,
20///  this processor is run in two passes:
21///   First, all nodes are run through and indices of pin changes are marked.
22///   Second, all non-PinAction nodes are copied over, with only PinAction nodes whose indices were marked are copied over.
23impl PinActionCombiner {
24    pub fn run(node: &Node<PAT>) -> Result<Node<PAT>> {
25        let mut p = PinActionCombiner {
26            current_state: HashMap::new(),
27            i: 0,
28            first_pass: true,
29            delete_current_index: false,
30            indices_to_delete: vec![],
31        };
32
33        node.process(&mut p)?.unwrap();
34        p.advance_to_second_pass();
35        let n = node.process(&mut p)?.unwrap();
36        Ok(n)
37    }
38
39    pub fn advance_to_second_pass(&mut self) {
40        self.i = 0;
41        self.first_pass = false;
42    }
43}
44
45impl Processor<PAT> for PinActionCombiner {
46    fn on_node(&mut self, node: &Node<PAT>) -> origen_metal::Result<Return<PAT>> {
47        match &node.attrs {
48            // Grab the pins and push them into the running vectors, then delete this node.
49            PAT::PinGroupAction(_grp_id, _actions, _metadata) => {
50                if self.first_pass {
51                    // On the first pass, only mark the pin group's index
52                    self.delete_current_index = true;
53                    Ok(Return::ProcessChildren)
54                } else {
55                    // On the second pass, delete the pin group if necessary
56                    if self.indices_to_delete.contains(&self.i) {
57                        self.i += 1;
58                        Ok(Return::None)
59                    } else {
60                        self.i += 1;
61                        Ok(Return::Unmodified)
62                    }
63                }
64            }
65            PAT::PinAction(pin_id, action, _metadata) => {
66                if self.first_pass {
67                    // Compare to the last seen pin actions. If these are the same, then this node can be deleted on the next pass.
68                    // If they're different, then these updates must be saved.
69                    if let Some(a) = self.current_state.get_mut(pin_id) {
70                        if a != action {
71                            self.delete_current_index = false;
72                            self.current_state.insert(*pin_id, action.clone());
73                        }
74                    } else {
75                        self.current_state.insert(*pin_id, action.clone());
76                        self.delete_current_index = false;
77                    }
78                }
79                Ok(Return::Unmodified)
80            }
81            _ => Ok(Return::ProcessChildren),
82        }
83    }
84
85    fn on_processed_node(&mut self, node: &Node<PAT>) -> origen_metal::Result<Return<PAT>> {
86        match &node.attrs {
87            PAT::PinGroupAction(_grp_id, _actions, _metadata) => {
88                if self.first_pass {
89                    if self.delete_current_index {
90                        self.indices_to_delete.push(self.i);
91                    }
92                    self.delete_current_index = false;
93                    self.i += 1;
94                }
95                Ok(Return::Unmodified)
96            }
97            _ => Ok(Return::Unmodified),
98        }
99    }
100}
