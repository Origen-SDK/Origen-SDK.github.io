1use crate::proc_macro::TokenStream;
2use quote::{format_ident, quote};
3use syn;
4
5#[derive(Default)]
6struct Config {
7    pub getter_type: String,
8    pub field: String,
9    pub field_container_name: String,
10    pub parent_field: String,
11    pub return_type: String,
12}
13
14pub fn impl_id_getters(ast: &syn::DeriveInput) -> syn::Result<TokenStream> {
15    let name = &ast.ident;
16    let mut getter_functions = quote! {};
17    for (_i, attr) in ast.attrs.iter().enumerate() {
18        // Go through the given attributes. If something doesn't parse, just ignore it. The Rust language will eithe catch for something
19        // grossly malformed, or it may be a (very) customized attribute.
20        // Likewise, ignore any atttributes that aren't ours.
21        match attr.parse_meta()? {
22            syn::Meta::List(syn::MetaList {
23                ref path,
24                ref nested,
25                ..
26            }) => {
27                if path.is_ident("id_getters_by_index") || path.is_ident("id_getters_by_mapping") {
28                    let mut config = Config::default();
29                    if path.is_ident("id_getters_by_index") {
30                        config.getter_type = "by_index".to_string();
31                    } else {
32                        config.getter_type = "by_mapping".to_string();
33                    }
34                    for (_i, _attr) in nested.iter().enumerate() {
35                        match _attr {
36                            syn::NestedMeta::Meta(syn::Meta::NameValue(ref name_value_pair)) => {
37                                // Match the name/value pair to the appropriate field in the config.
38                                if name_value_pair.path.is_ident("field") {
39                                    match &name_value_pair.lit {
40                                    syn::Lit::Str(l) => config.field = l.value(),
41                                    _ => return Err(syn::Error::new_spanned(name_value_pair.lit.clone(), format!("Could not process name-value pair's value as a String")))
42                                }
43                                } else if name_value_pair.path.is_ident("parent_field") {
44                                    match &name_value_pair.lit {
45                                    syn::Lit::Str(l) => config.parent_field = l.value(),
46                                    _ => return Err(syn::Error::new_spanned(name_value_pair.lit.clone(), format!("Could not process name-value pair's value as a String")))
47                                }
48                                } else if name_value_pair.path.is_ident("field_container_name") {
49                                    match &name_value_pair.lit {
50                                    syn::Lit::Str(l) => config.field_container_name = l.value(),
51                                    _ => return Err(syn::Error::new_spanned(name_value_pair.lit.clone(), format!("Could not process name-value pair's value as a String")))
52                                }
53                                } else if name_value_pair.path.is_ident("return_type") {
54                                    match &name_value_pair.lit {
55                                    syn::Lit::Str(l) => config.return_type = l.value(),
56                                    _ => return Err(syn::Error::new_spanned(name_value_pair.lit.clone(), format!("Could not process name-value pair's value as a String")))
57                                }
58                                } else {
59                                    return Err(syn::Error::new_spanned(
60                                        name_value_pair.path.clone(),
61                                        format!(
62                                            "Unexpected name-value pair '{}'",
63                                            name_value_pair.path.get_ident().unwrap()
64                                        ),
65                                    ));
66                                }
67                            }
68                            _ => {
69                                return Err(syn::Error::new_spanned(
70                                    _attr,
71                                    format!("Could not process as a name-value pair!"),
72                                ))
73                            }
74                        }
75                    }
76
77                    let func_name = format_ident!("get_{}", config.field);
78                    let mut_func_name = format_ident!("get_mut_{}", config.field);
79                    let _func_name = format_ident!("_{}", func_name);
80                    let _mut_func_name = format_ident!("_{}", mut_func_name);
81                    let clone_func_name = format_ident!("_get_cloned_{}", config.field);
82
83                    let get_id_func = format_ident!("get_{}_id", config.field);
84                    let retn = format_ident!("{}", config.return_type);
85                    let field_container_name = format_ident!("{}", config.field_container_name);
86                    let parent_field = format_ident!("{}", config.parent_field);
87
88                    let (lookup_type, error_message, err_str);
89                    if config.getter_type == "by_index" {
90                        lookup_type = quote! { usize };
91                        err_str = format!("\"Could not find {} at index {{}}!\"", config.field);
92                    } else {
93                        lookup_type = quote! { &str };
94                        err_str = format!("\"Could not find {} named {{}}!\"", config.field);
95                    }
96                    error_message = quote! { &format!(#err_str, identifier) };
97
98                    getter_functions.extend(quote! {
99                    impl #name {
100                        pub fn #func_name (&self, parent_field_id: usize, identifier: #lookup_type) -> Option<& #retn > {
101                            if let Some(i) = self.#parent_field[parent_field_id].#get_id_func(identifier) {
102                                Some(&self.#field_container_name[i])
103                            } else {
104                                Option::None
105                            }
106                        }
107
108                        pub fn #mut_func_name(&mut self, parent_field_id: usize, identifier: #lookup_type) -> Option<&mut #retn> {
109                            if let Some(i) = self.#parent_field[parent_field_id].#get_id_func(identifier) {
110                                Some(&mut self.#field_container_name[i])
111                            } else {
112                                Option::None
113                            }
114                        }
115
116                        pub fn #_func_name (&self, parent_field_id: usize, identifier: #lookup_type) -> core::result::Result<& #retn, crate::Error> {
117                            if let Some(i) = self.#parent_field[parent_field_id].#get_id_func(identifier) {
118                                Ok(&self.#field_container_name[i])
119                            } else {
120                                Err(crate::Error::new(#error_message))
121                            }
122                        }
123
124                        pub fn #_mut_func_name(&mut self, parent_field_id: usize, identifier: #lookup_type) -> core::result::Result<&mut #retn, crate::Error> {
125                            if let Some(i) = self.#parent_field[parent_field_id].#get_id_func(identifier) {
126                                Ok(&mut self.#field_container_name[i])
127                            } else {
128                                Err(crate::Error::new(#error_message))
129                            }
130                        }
131
132                        pub fn #clone_func_name(&self, parent_field_id: usize, identifier: #lookup_type) -> core::result::Result<#retn, crate::Error> {
133                            if let Some(i) = self.#parent_field[parent_field_id].#get_id_func(identifier) {
134                                Ok(self.#field_container_name[i].clone())
135                            } else {
136                                Err(crate::Error::new(#error_message))
137                            }
138                        }
139                    }
140                });
141                }
142            }
143            _ => {}
144        }
145    }
146    Ok(getter_functions.into())
147}
