1use crate::ast::{Node, Processor, Return};
2use crate::prog_gen::{Bin, BinType, FlowCondition, Model, VariableOperation, VariableType, PGM};
3use crate::Result;
4
5/// This extracts all definitions for tests, test invocations, pattern sets, bins, etc.
6/// and converts them into a program model which is returned.
7/// The resultant AST has most of the associated nodes removed but is otherwise unchanged.
8/// The model is not considered finalized until after the flow generator for the specific ATE
9/// target has run, at that point any ATE-specific extraction into the model will be complete,
10/// e.g. to extract pattern references made by test objects.
11pub struct ExtractToModel {
12    model: Model,
13}
14
15pub fn run(node: &Node<PGM>, model: Model) -> Result<(Node<PGM>, Model)> {
16    let mut p = ExtractToModel { model: model };
17    let ast = node.process(&mut p)?.unwrap();
18    Ok((ast, p.model))
19}
20
21impl Processor<PGM> for ExtractToModel {
22    fn on_node(&mut self, node: &Node<PGM>) -> crate::Result<Return<PGM>> {
23        Ok(match &node.attrs {
24            PGM::ResourcesFilename(name, kind) => {
25                self.model.set_resources_filename(name.to_owned(), kind);
26                Return::Unmodified
27            }
28            PGM::Condition(cond) => {
29                match cond {
30                    FlowCondition::IfJob(_ids) | FlowCondition::UnlessJob(_ids) => {
31                        self.model.record_variable_reference(
32                            "JOB".to_string(),
33                            VariableType::Job,
34                            VariableOperation::Reference,
35                        );
36                    }
37                    FlowCondition::IfEnable(ids) | FlowCondition::UnlessEnable(ids) => {
38                        for id in ids {
39                            self.model.record_variable_reference(
40                                id.to_owned(),
41                                VariableType::Enable,
42                                VariableOperation::Reference,
43                            );
44                        }
45                    }
46                    FlowCondition::IfFlag(ids) | FlowCondition::UnlessFlag(ids) => {
47                        for id in ids {
48                            self.model.record_variable_reference(
49                                id.to_owned(),
50                                VariableType::Flag,
51                                VariableOperation::Reference,
52                            );
53                        }
54                    }
55                    _ => {}
56                }
57                Return::ProcessChildren
58            }
59            PGM::SetFlag(name, _, _) => {
60                self.model.record_variable_reference(
61                    name.to_owned(),
62                    VariableType::Flag,
63                    VariableOperation::Set,
64                );
65                Return::ProcessChildren
66            }
67            //PGM::PatternGroup(id, name, _, kind) => Ok(Return::None),
68            //PGM::PushPattern(id, name, start_label) => Ok(Return::None),
69            // These will be left in the AST for later consumption by the flow, but they also function
70            // as un-official bin definitions.
71            // Any official bin defs in the AST for the same bin number will be combined with the model
72            // defintion created from here.
73            PGM::Bin(hardbin, softbin, kind) => {
74                let flow = self.model.get_flow_mut(None);
75                if !flow.hardbins.contains_key(hardbin) {
76                    let b = Bin {
77                        number: *hardbin,
78                        description: None,
79                        priority: None,
80                        pass: matches!(kind, BinType::Good),
81                    };
82                    flow.hardbins.insert(*hardbin, b);
83                }
84                if let Some(softbin) = softbin {
85                    if !flow.softbins.contains_key(hardbin) {
86                        let b = Bin {
87                            number: *softbin,
88                            description: None,
89                            priority: None,
90                            pass: matches!(kind, BinType::Good),
91                        };
92                        flow.softbins.insert(*softbin, b);
93                    }
94                }
95                Return::Unmodified
96            }
97            PGM::DefBin(number, is_soft, kind, description, priority) => {
98                let flow = self.model.get_flow_mut(None);
99                let collection = match is_soft {
100                    false => &mut flow.hardbins,
101                    true => &mut flow.softbins,
102                };
103                if !collection.contains_key(number) {
104                    let b = Bin {
105                        number: *number,
106                        description: description.to_owned(),
107                        priority: priority.to_owned(),
108                        pass: matches!(kind, BinType::Good),
109                    };
110                    collection.insert(*number, b);
111                } else {
112                    let b = collection.get_mut(number).unwrap();
113                    if let Some(d) = description {
114                        b.description = Some(d.to_owned());
115                    }
116                    if let Some(p) = priority {
117                        b.priority = Some(*p);
118                    }
119                }
120                Return::None
121            }
122            _ => Return::ProcessChildren,
123        })
124    }
125}
