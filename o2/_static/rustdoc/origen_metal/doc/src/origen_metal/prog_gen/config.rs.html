5/// Configuration for the program generator, an singleton is instantiated as
6/// `PROG_GEN_CONFIG` in `lib.rs`
7#[derive(Debug)]
8pub struct Config {
9    app_name: RwLock<Option<String>>,
10    unique_id: RwLock<usize>,
11    debug_enabled: RwLock<bool>,
12    src_files: RwLock<Vec<PathBuf>>,
13    /// The test template load path, with a revision which advances each time it is changed
14    test_template_load_path: RwLock<(usize, Vec<PathBuf>)>,
15    flow_visualization_enabled: RwLock<bool>,
16    uniqueness_option: RwLock<Option<UniquenessOption>>,
17    smt7: RwLock<SMT7Config>,
18    smt8: RwLock<SMT8Config>,
19    /// Render an empty else branch of a V93K SMT8 condition
20    smt8_empty_else: RwLock<bool>,
21    /// Format V93K SMT8 flows as Origen 1 (origen_testers) writes them
22    smt8_o1_formatting: RwLock<bool>,
23    /// Keep the case of V93K SMT8 sub-flow instance names
24    smt8_keep_sub_flow_instance_case: RwLock<bool>,
25}

103    pub fn set_test_template_load_path(&self, paths: Vec<PathBuf>) {

171    /// Set whether V93K SMT8 flows are formatted as Origen 1 (origen_testers) writes them, which is

202#[cfg(test)]
203mod tests {
204    use super::*;
205
206    #[test]
207    fn the_load_path_is_kept_as_given_and_its_revision_advances_when_it_changes() {
208        let config = Config::default();
209        config.set_test_template_load_path(vec![PathBuf::from("templates//version")]);
210        let (revision, _) = config.test_template_load_path_revision();
211        // The same directory, spelled differently
212        config.set_test_template_load_path(vec![PathBuf::from("templates/version")]);
213        let (same_revision, paths) = config.test_template_load_path_revision();
214        assert_eq!(same_revision, revision);
215        assert_eq!(paths[0].as_os_str(), "templates/version");
216        assert_eq!(
217            config.test_template_load_path()[0].as_os_str(),
218            "templates/version"
219        );
220        config.set_test_template_load_path(vec![PathBuf::from("other")]);
221        assert_eq!(config.test_template_load_path_revision().0, revision + 1);
222    }
