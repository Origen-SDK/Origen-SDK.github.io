4/// Configuration for the program generator, an singleton is instantiated as
5/// `PROG_GEN_CONFIG` in `lib.rs`
6#[derive(Debug)]
7pub struct Config {
8    app_name: RwLock<Option<String>>,
9    unique_id: RwLock<usize>,
10    debug_enabled: RwLock<bool>,
11    src_files: RwLock<Vec<PathBuf>>,
12    test_template_load_path: RwLock<Vec<PathBuf>>,
13    uniqueness_option: RwLock<Option<UniquenessOption>>,
14    smt7: RwLock<SMT7Config>,
15    smt8: RwLock<SMT8Config>,
16}
17
18#[derive(Debug, Clone)]
19pub struct SMT8Config {
20    pub create_limits_file: bool,
21    pub render_default_tmparams: bool,
22}
23
24#[derive(Debug, Clone)]
25pub struct SMT7Config {
26    pub render_default_tmparams: bool,
27}
28
29impl Default for Config {
30    fn default() -> Self {
31        Config {
32            app_name: RwLock::new(None),
33            unique_id: RwLock::new(0),
34            debug_enabled: RwLock::new(false),
35            src_files: RwLock::new(vec![]),
36            test_template_load_path: RwLock::new(vec![]),
37            uniqueness_option: RwLock::new(None),
38            smt7: RwLock::new(SMT7Config {
39                render_default_tmparams: true,
40            }),
41            smt8: RwLock::new(SMT8Config {
42                create_limits_file: true,
43                render_default_tmparams: true,
44            }),
45        }
46    }
47}

49impl Config {
50    pub fn debug_enabled(&self) -> bool {
51        *self.debug_enabled.read().unwrap()
52    }
53
54    pub fn set_debug_enabled(&self, enabled: bool) {
55        *self.debug_enabled.write().unwrap() = enabled;
56    }
57
58    pub fn app_name(&self) -> Option<String> {
59        self.app_name.read().unwrap().clone()
60    }
61
62    pub fn set_app_name(&self, name: String) {
63        *self.app_name.write().unwrap() = Some(name);
64    }
65
66    /// Returns an ID that it guaranteed unique across threads and within the lifetime of an Origen
67    /// invocation
68    pub fn generate_unique_id(&self) -> usize {
69        let mut unique_id = self.unique_id.write().unwrap();
70        *unique_id += 1;
71        *unique_id
72    }
73
74    pub fn start_src_file(&self, file: PathBuf) -> crate::Result<()> {
75        let mut src_files = self.src_files.write().unwrap();
76        src_files.push(file);
77        Ok(())
78    }
79
80    pub fn end_src_file(&self) {
81        let mut src_files = self.src_files.write().unwrap();
82        src_files.pop();
83    }
84
85    pub fn current_src_file(&self) -> Option<PathBuf> {
86        let src_files = self.src_files.read().unwrap();
87        src_files.last().cloned()
88    }
89
90    pub fn set_test_template_load_path(&self, paths: Vec<PathBuf>) {
91        let mut lp = self.test_template_load_path.write().unwrap();
92        *lp = paths;
93    }
94
95    pub fn test_template_load_path(&self) -> Vec<PathBuf> {
96        self.test_template_load_path.read().unwrap().clone()
97    }
98
99    pub fn set_uniqueness_option(&self, option: UniquenessOption) {
100        *self.uniqueness_option.write().unwrap() = Some(option);
101    }
102
103    pub fn uniqueness_option(&self) -> Option<UniquenessOption> {
104        self.uniqueness_option.read().unwrap().clone()
105    }

107    pub fn set_smt7_options(&self, render_default_tmparams: bool) {
108        *self.smt7.write().unwrap() = SMT7Config {

110        };
111    }
112
113    pub fn smt7_options(&self) -> SMT7Config {

115    }

117    pub fn set_smt8_options(&self, create_limits_file: bool, render_default_tmparams: bool) {

120            render_default_tmparams,

122    }

125        self.smt8.read().unwrap().clone()
