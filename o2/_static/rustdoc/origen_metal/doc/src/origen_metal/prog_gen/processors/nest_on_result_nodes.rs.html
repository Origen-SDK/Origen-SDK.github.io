6/// This nests OnFailed and OnPassed nodes into their parent tests (or groups)
7///
8/// Input:
9///     PGMTest(4, FlowID { id: "3" })
10///     PGMOnFailed(FlowID { id: "3" })
11///         PGMBin(100, Some(1100), Bad)
12///     PGMTest(6, FlowID { id: "7" })
13///     PGMOnFailed(FlowID { id: "7" })
14///         PGMBin(100, Some(1100), Bad)
15///
16/// Output:
17///     PGMTest(4, FlowID { id: "3" })
18///         PGMOnFailed(FlowID { id: "3" })
19///             PGMBin(100, Some(1100), Bad)
20///     PGMTest(6, FlowID { id: "7" })
21///         PGMOnFailed(FlowID { id: "7" })
22///             PGMBin(100, Some(1100), Bad)
23pub struct NestOnResultNodes {
24    nodes: IndexMap<FlowID, Node<PGM>>,
25    pass: usize,
26}
27
28pub fn run(node: &Node<PGM>) -> Result<Node<PGM>> {
29    let mut p = NestOnResultNodes {
30        nodes: IndexMap::new(),
31        pass: 0,
32    };
33    // The first pass strips out all OnPassed and OnFailed nodes and holds them
34    let ast = node.process(&mut p)?.unwrap();
35    p.pass = 1;
36    // Then the final pass re-inserts them in the right place
37    let ast = ast.process(&mut p)?.unwrap();
38    Ok(ast)
39}
40
41impl Processor<PGM> for NestOnResultNodes {
42    fn on_node(&mut self, node: &Node<PGM>) -> crate::Result<Return<PGM>> {
43        match &node.attrs {
44            PGM::OnFailed(fid) | PGM::OnPassed(fid) => {
45                if self.pass == 0 {
46                    let n = { node.process_and_update_children(self)? };
47                    self.nodes.insert(fid.to_owned(), n);
48                    Ok(Return::None)
49                } else {
50                    Ok(Return::ProcessChildren)
51                }
52            }
53            PGM::Test(_, fid) | PGM::TestStr(_, fid, _, _, _) | PGM::Cz(_, _, fid) => {
54                if self.pass == 1 && self.nodes.contains_key(fid) {
55                    let n = { self.nodes.shift_remove(fid).unwrap() };
56                    let mut nodes = vec![n.process_and_update_children(self)?];
57                    nodes.append(&mut node.process_children(self)?);
58                    Ok(Return::ReplaceChildren(nodes))
59                } else {
60                    Ok(Return::ProcessChildren)
61                }
62            }
63            PGM::Group(_, _, _, fid) => {
64                if let Some(fid) = fid {
65                    if self.pass == 1 && self.nodes.contains_key(fid) {
66                        let n = { self.nodes.shift_remove(fid).unwrap() };
67                        let mut nodes = vec![n.process_and_update_children(self)?];
68                        nodes.append(&mut node.process_children(self)?);
69                        Ok(Return::ReplaceChildren(nodes))
70                    } else {
71                        Ok(Return::ProcessChildren)
72                    }
73                } else {
74                    Ok(Return::ProcessChildren)
75                }
76            }
77            _ => Ok(Return::ProcessChildren),
78        }
79    }
80}

82#[cfg(test)]
83mod tests {
84    use super::run;
85    use crate::prog_gen::{BinType, FlowID, PGM};
86    use crate::Result;
87
88    #[test]
89    fn it_works() -> Result<()> {
90        let input = node!(PGM::Flow, "f1".to_string() =>
91            node!(PGM::Test, 1, FlowID::from_int(1)),
92            node!(PGM::Test, 2, FlowID::from_int(2)),
93            node!(PGM::OnFailed, FlowID::from_int(2) =>
94                node!(PGM::Bin, 10, None, BinType::Bad)
95            ),
96            node!(PGM::Test, 3, FlowID::from_int(3)),
97            node!(PGM::OnPassed, FlowID::from_int(3) =>
98                node!(PGM::Test, 4, FlowID::from_int(4)),
99                node!(PGM::OnFailed, FlowID::from_int(4) =>
100                    node!(PGM::Bin, 10, None, BinType::Bad)
101                )
102            )
103        );
104
105        let expected = node!(PGM::Flow, "f1".to_string() =>
106            node!(PGM::Test, 1, FlowID::from_int(1)),
107            node!(PGM::Test, 2, FlowID::from_int(2) =>
108                node!(PGM::OnFailed, FlowID::from_int(2) =>
109                    node!(PGM::Bin, 10, None, BinType::Bad)
110                )
111            ),
112            node!(PGM::Test, 3, FlowID::from_int(3) =>
113                node!(PGM::OnPassed, FlowID::from_int(3) =>
114                    node!(PGM::Test, 4, FlowID::from_int(4) =>
115                        node!(PGM::OnFailed, FlowID::from_int(4) =>
116                            node!(PGM::Bin, 10, None, BinType::Bad)
117                        )
118                    )
119                )
120            )
121        );
122
123        assert_eq!(run(&input)?, expected);
124        Ok(())
125    }
