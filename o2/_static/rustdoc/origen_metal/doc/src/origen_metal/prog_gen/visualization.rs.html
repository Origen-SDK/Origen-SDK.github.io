75    pub fn from_processed_ast(
76        name: &str,
77        tester: SupportedTester,
78        ast: &Node<PGM>,
79        model: &Model,
80    ) -> Self {
81        let mut builder = GraphBuilder {
82            graph: FlowGraph {
83                schema_version: SCHEMA_VERSION,
84                name: name.to_string(),
85                tester: tester.to_string(),
86                nodes: Vec::new(),
87                edges: Vec::new(),
88                tests: IndexMap::new(),
89            },
90            model,
91            next_node: 0,
92        };
93        if matches!(ast.attrs, PGM::Flow(_)) {
94            builder.build_sequence(&ast.children, 0);
95        } else {
96            builder.build_node(ast, 0);
97        }
98        builder.graph
99    }

102impl<'a> GraphBuilder<'a> {
103    fn build_sequence(&mut self, nodes: &[Box<Node<PGM>>], depth: usize) -> SequenceResult {
104        let mut result = SequenceResult::default();
105        let mut previous_exits: Vec<String> = Vec::new();
106
107        for node in nodes {
108            let current = self.build_node(node, depth);
109            let Some(entry) = current.entry.clone() else {
110                continue;
111            };
112            if result.entry.is_none() {
113                result.entry = Some(entry.clone());
114            }
115            for previous in &previous_exits {
116                self.push_edge(previous, &entry, "next", None);
117            }
118            previous_exits = current.exits;
119        }
120        result.exits = previous_exits;
121        result
122    }
123
124    fn build_node(&mut self, node: &Node<PGM>, depth: usize) -> SequenceResult {
125        let Some((kind, label, test_id)) = describe_node(&node.attrs, self.model) else {
126            return self.build_sequence(&node.children, depth);
127        };
128
129        let id = format!("n{}", self.next_node);
130        self.next_node += 1;
131        let detail_id = test_id.map(|test_id| self.add_test_details(test_id));
132        self.graph.nodes.push(FlowGraphNode {
133            id: id.clone(),
134            kind: kind.to_string(),
135            label,
136            detail_id,
137            source: node.meta.as_ref().and_then(|meta| {
138                meta.filename.as_ref().map(|filename| match meta.lineno {
139                    Some(line) => format!("{}:{}", filename, line),
140                    None => filename.clone(),
141                })
142            }),
143            depth,
144            order: self.graph.nodes.len(),
145        });
146
147        if node.children.is_empty() {
148            return SequenceResult {
149                entry: Some(id.clone()),
150                exits: if is_terminal(&node.attrs) {
151                    Vec::new()
152                } else {
153                    vec![id]
154                },
155            };
156        }
157
158        if matches!(node.attrs, PGM::Test(_, _) | PGM::Cz(_, _, _)) {
159            let mut exits = vec![id.clone()];
160            for child in &node.children {
161                if let Some(branch_kind) = result_handler_kind(&child.attrs) {
162                    let branch = self.build_node(child, depth + 1);
163                    if let Some(entry) = branch.entry {
164                        self.push_edge(&id, &entry, branch_kind, Some(branch_kind));
165                    }
166                    exits.extend(branch.exits);
167                } else {
168                    let branch = self.build_node(child, depth + 1);
169                    if let Some(entry) = branch.entry {
170                        self.push_edge(&id, &entry, "child", None);
171                    }
172                    exits.extend(branch.exits);
173                }
174            }
175            return SequenceResult {
176                entry: Some(id),
177                exits,
178            };
179        }
180
181        let children = self.build_sequence(&node.children, depth + 1);
182        if let Some(entry) = children.entry {
183            let label = block_edge_label(&node.attrs);
184            self.push_edge(&id, &entry, block_edge_kind(&node.attrs), label);
185        }
186
187        let mut exits = children.exits;
188        if has_fallthrough(&node.attrs) {
189            exits.push(id.clone());
190        }
191        if matches!(node.attrs, PGM::Loop(_, _)) {
192            for exit in &exits {
193                if exit != &id {
194                    self.push_edge(exit, &id, "loop", Some("repeat"));
195                }
196            }
197            exits = vec![id.clone()];
198        }
199        if exits.is_empty() && !is_terminal(&node.attrs) {
200            exits.push(id.clone());
201        }
202        SequenceResult {
203            entry: Some(id),
204            exits,
205        }
206    }

208    fn add_test_details(&mut self, id: usize) -> String {
209        let detail_id = format!("test-{}", id);
210        if self.graph.tests.contains_key(&detail_id) {
211            return detail_id;
212        }
213
214        let invocation = self.model.test_invocations.get(&id);
215        let method = invocation
216            .and_then(|invocation| invocation.test_id)
217            .and_then(|test_id| self.model.tests.get(&test_id))
218            .or_else(|| self.model.tests.get(&id));
219
220        let details = match method {
221            Some(method) => {
222                let mut parameters = parameter_details("method", "", method);
223                if let Some(invocation) = invocation {
224                    parameters.extend(parameter_details("invocation", "", invocation));
225                }
226                TestDetails {
227                    invocation: invocation.map(|invocation| invocation.name.clone()),
228                    method: method.name.clone(),
229                    library: method.template_library.clone(),
230                    template: method.template_name.clone(),
231                    class_name: method.class_name.clone(),
232                    parameters,
233                }
234            }
235            None => TestDetails {
236                invocation: None,
237                method: format!("Test {}", id),
238                library: None,
239                template: None,
240                class_name: None,
241                parameters: Vec::new(),
242            },
243        };
244        self.graph.tests.insert(detail_id.clone(), details);
245        detail_id
246    }
247
248    fn push_edge(&mut self, from: &str, to: &str, kind: &str, label: Option<&str>) {
249        if from == to
250            || self
251                .graph
252                .edges
253                .iter()
254                .any(|edge| edge.from == from && edge.to == to && edge.kind == kind)
255        {
256            return;
257        }
258        self.graph.edges.push(FlowGraphEdge {
259            from: from.to_string(),
260            to: to.to_string(),
261            kind: kind.to_string(),
262            label: label.map(str::to_string),
263        });
264    }
265}

267fn parameter_details(scope: &str, prefix: &str, test: &Test) -> Vec<ParameterDetails> {
268    let mut parameters = Vec::new();
269    for (name, kind) in &test.params {
270        let value = test.values.get(name);
271        let default = test.default_values.get(name);
272        let configured = match (value, default) {
273            (Some(value), Some(default)) => value != default,
274            (Some(_), None) => true,
275            _ => false,
276        };
277        parameters.push(ParameterDetails {
278            scope: scope.to_string(),
279            path: join_path(prefix, name),
280            kind: parameter_kind(kind),
281            value: if configured {
282                value.map(parameter_value)
283            } else {
284                None
285            },
286            default: default.map(parameter_value),
287            configured,
288            aliases: aliases_for(name, &test.aliases),
289            constraints: test
290                .constraints
291                .get(name)
292                .map(|constraints| {
293                    constraints
294                        .iter()
295                        .map(|constraint| format!("{:?}", constraint))
296                        .collect()
297                })
298                .unwrap_or_default(),
299        });
300    }
301    for (collection_name, collection) in &test.collection_defs {
302        collect_collection_parameters(
303            scope,
304            &join_path(prefix, collection_name),
305            collection,
306            &mut parameters,
307        );
308    }
309    parameters.sort_by(|a, b| a.scope.cmp(&b.scope).then(a.path.cmp(&b.path)));
310    parameters
311}

313fn collect_collection_parameters(
314    scope: &str,
315    prefix: &str,
316    collection: &super::TestCollection,
317    parameters: &mut Vec<ParameterDetails>,
318) {
319    for (name, kind) in &collection.params {
320        parameters.push(ParameterDetails {
321            scope: scope.to_string(),
322            path: join_path(prefix, name),
323            kind: parameter_kind(kind),
324            value: None,
325            default: collection.default_values.get(name).map(parameter_value),
326            configured: false,
327            aliases: aliases_for(name, &collection.aliases),
328            constraints: collection
329                .constraints
330                .get(name)
331                .map(|constraints| {
332                    constraints
333                        .iter()
334                        .map(|constraint| format!("{:?}", constraint))
335                        .collect()
336                })
337                .unwrap_or_default(),
338        });
339    }
340    for (name, nested) in &collection.collections {
341        collect_collection_parameters(scope, &join_path(prefix, name), nested, parameters);
342    }
343}

345fn aliases_for(name: &str, aliases: &IndexMap<String, String>) -> Vec<String> {
346    let mut result: Vec<String> = aliases
347        .iter()
348        .filter_map(|(alias, target)| {
349            if target == name {
350                Some(alias.clone())
351            } else {
352                None
353            }
354        })
355        .collect();
356    result.sort();
357    result
358}

372fn parameter_value(value: &ParamValue) -> String {
373    match value {
374        ParamValue::String(value) | ParamValue::Class(value) | ParamValue::Any(value) => {
375            value.clone()
376        }
377        ParamValue::Bool(value) => value.to_string(),
378        ParamValue::Int(value) => value.to_string(),
379        ParamValue::UInt(value) => value.to_string(),
380        ParamValue::Float(value) => value.to_string(),
381        ParamValue::Current(value) => format!("{} A", value),
382        ParamValue::Voltage(value) => format!("{} V", value),
383        ParamValue::Time(value) => format!("{} s", value),
384        ParamValue::Frequency(value) => format!("{} Hz", value),
385        ParamValue::List(values) => values
386            .iter()
387            .map(parameter_value)
388            .collect::<Vec<_>>()
389            .join(", "),
390    }

393fn describe_node(attrs: &PGM, model: &Model) -> Option<(&'static str, String, Option<usize>)> {
394    Some(match attrs {
395        PGM::Nil
396        | PGM::DefTest(_, _, _, _, _)
397        | PGM::DefTestInv(_, _, _)
398        | PGM::AssignTestToInv(_, _)
399        | PGM::DefTestCollectionItem(_, _, _, _, _)
400        | PGM::SetAttr(_, _, _, _)
401        | PGM::SetLimit(_, _, _, _)
402        | PGM::DefSubTest(_, _, _, _, _)
403        | PGM::PatternGroup(_, _, _, _)
404        | PGM::PushPattern(_, _, _)
405        | PGM::DefBin(_, _, _, _, _)
406        | PGM::Resources
407        | PGM::ResourcesFilename(_, _)
408        | PGM::FlowDescription(_)
409        | PGM::FlowNameOverride(_)
410        | PGM::Namespace(_)
411        | PGM::Uniqueness(_)
412        | PGM::IGXLResource(_)
413        | PGM::IGXLResourcesFilename(_, _)
414        | PGM::Variable(_, _, _)
415        | PGM::Parameter(_, _, _) => return None,
416        PGM::Flow(name) => ("flow", name.clone(), None),
417        PGM::SubFlow(name, _) => ("subflow", format!("Subflow: {}", name), None),
418        PGM::Test(id, flow_id) => {
419            let name = model
420                .test_invocations
421                .get(id)
422                .or_else(|| model.tests.get(id))
423                .map(|test| test.name.clone())
424                .unwrap_or_else(|| format!("Test {}", id));
425            ("test", format!("{}\n{}", name, flow_id.to_str()), Some(*id))
426        }
427        PGM::TestStr(name, flow_id, _, _, _) => {
428            ("test", format!("{}\n{}", name, flow_id.to_str()), None)
429        }
430        PGM::Cz(id, setup, flow_id) => {
431            let name = model
432                .test_invocations
433                .get(id)
434                .or_else(|| model.tests.get(id))
435                .map(|test| test.name.clone())
436                .unwrap_or_else(|| format!("Test {}", id));
437            (
438                "test",
439                format!("{}\nCZ: {}\n{}", name, setup, flow_id.to_str()),
440                Some(*id),
441            )
442        }
443        PGM::Render(text) => ("action", truncate(text, 72), None),
444        PGM::Log(text) => ("action", format!("Log: {}", truncate(text, 64)), None),
445        PGM::Group(name, _, kind, _) => ("group", format!("{:?} group: {}", kind, name), None),
446        PGM::Condition(condition) => ("condition", condition_label(condition), None),
447        PGM::Bin(number, soft, kind) => (
448            match kind {
449                BinType::Good => "bin-good",
450                BinType::Bad => "bin-bad",
451            },
452            match soft {
453                Some(soft) => format!("{:?} bin {} / soft {}", kind, number, soft),
454                None => format!("{:?} bin {}", kind, number),
455            },
456            None,
457        ),
458        PGM::OnFailed(_) => ("handler-fail", "On fail".to_string(), None),
459        PGM::OnPassed(_) => ("handler-pass", "On pass".to_string(), None),
460        PGM::OnError(_) => ("handler-error", "On error".to_string(), None),
461        PGM::TesterEq(testers) => ("condition", format!("Tester is {:?}", testers), None),
462        PGM::TesterNeq(testers) => ("condition", format!("Tester is not {:?}", testers), None),
463        PGM::Volatile(flag) => ("action", format!("Volatile flag: {}", flag), None),
464        PGM::SetFlag(flag, value, _) => ("action", format!("Set {} = {}", flag, value), None),
465        PGM::SetDefaultFlagState(flag, value) => {
466            ("action", format!("Default {} = {}", flag, value), None)
467        }
468        PGM::Continue => ("action", "Continue on failure".to_string(), None),
469        PGM::Delayed => ("action", "Delay binning".to_string(), None),
470        PGM::Else => ("condition", "Else".to_string(), None),
471        PGM::Whenever => ("condition", "Whenever".to_string(), None),
472        PGM::WheneverAny => ("condition", "Whenever any".to_string(), None),
473        PGM::WheneverAll => ("condition", "Whenever all".to_string(), None),
474        PGM::Enable(flag) => ("action", format!("Enable {}", flag), None),
475        PGM::Disable(flag) => ("action", format!("Disable {}", flag), None),
476        PGM::BypassSubFlows => ("action", "Bypass subflows".to_string(), None),
477        PGM::IGXLSetWaitFlags(id, flags) => (
478            "action",
479            format!("Wait flags {}: {}", id, flags.join(", ")),
480            None,
481        ),
482        PGM::FlowData(_) => ("action", "Tester flow data".to_string(), None),
483        PGM::Unknown(kind, raw) => ("action", format!("{}: {}", kind, truncate(raw, 56)), None),
484        PGM::Comment(text) => ("note", truncate(text, 72), None),
485        PGM::Wait(duration) => ("action", format!("Wait {}", duration), None),
486        PGM::SetVariable(name, value) => ("action", format!("Set {} = {}", name, value), None),
487        PGM::Synchronize => ("action", "Synchronize sites".to_string(), None),
488        PGM::Label(label) => ("label", format!("Label: {}", label), None),
489        PGM::Goto(label) => ("action", format!("Go to {}", label), None),
490        PGM::Call(name, args) => (
491            "action",
492            format!("Call {}({})", name, args.join(", ")),
493            None,
494        ),
495        PGM::Loop(count, variable) => (
496            "condition",
497            format!(
498                "Loop {}{}",
499                count
500                    .map(|count| count.to_string())
501                    .unwrap_or_else(|| "until complete".to_string()),
502                variable
503                    .as_ref()
504                    .map(|variable| format!(" as {}", variable))
505                    .unwrap_or_default()
506            ),
507            None,
508        ),
509        PGM::Report(category, message) => (
510            "action",
511            format!("Report {}: {}", category, truncate(message, 56)),
512            None,
513        ),
514        PGM::Assertion(expression, message) => (
515            "condition",
516            format!("Assert {}: {}", expression, truncate(message, 48)),
517            None,
518        ),
519        PGM::Callback(name, args) => (
520            "action",
521            format!("Callback {}({})", name, args.join(", ")),
522            None,
523        ),
524    })
525}

527fn condition_label(condition: &FlowCondition) -> String {
528    match condition {
529        FlowCondition::IfJob(values) => format!("If job: {}", values.join(", ")),
530        FlowCondition::UnlessJob(values) => format!("Unless job: {}", values.join(", ")),
531        FlowCondition::IfEnable(values) => format!("If enabled: {}", values.join(", ")),
532        FlowCondition::UnlessEnable(values) => format!("Unless enabled: {}", values.join(", ")),
533        FlowCondition::IfFlag(values) => format!("If flag: {}", values.join(", ")),
534        FlowCondition::UnlessFlag(values) => format!("Unless flag: {}", values.join(", ")),
535        FlowCondition::IfExpr(expression) => format!("If {}", expression),
536        FlowCondition::UnlessExpr(expression) => format!("Unless {}", expression),
537        FlowCondition::IfVar(name, operator, value) => {
538            format!("If {} {} {}", name, operator, value)
539        }
540        FlowCondition::UnlessVar(name, operator, value) => {
541            format!("Unless {} {} {}", name, operator, value)
542        }
543        FlowCondition::IfSite(sites) => format!("If site: {:?}", sites),
544        FlowCondition::UnlessSite(sites) => format!("Unless site: {:?}", sites),
545        _ => format!("{:?}", condition),
546    }
547}

596fn truncate(value: &str, maximum: usize) -> String {
597    let mut chars = value.chars();
598    let shortened: String = chars.by_ref().take(maximum).collect();
599    if chars.next().is_some() {
600        format!("{}...", shortened)
601    } else {
602        shortened
603    }
604}

606pub fn render_flow_visualization(
607    name: &str,
608    tester: SupportedTester,
609    ast: &Node<PGM>,
610    model: &Model,
611    output_dir: &Path,
612) -> Result<Vec<PathBuf>> {
613    let graph = FlowGraph::from_processed_ast(name, tester, ast, model);
614    let directory = output_dir.join("flow_visualizations");
615    fs::create_dir_all(&directory)?;
616    let stem = safe_filename(name);
617    let json_path = directory.join(format!("{}.flow.json", stem));
618    let html_path = directory.join(format!("{}.flow.html", stem));
619    let json = serde_json::to_string_pretty(&graph).map_err(|error| {
620        crate::Error::new(&format!("Failed to serialize flow graph: {}", error))
621    })?;
622    fs::write(&json_path, &json)?;
623    fs::write(
624        &html_path,
625        FLOW_HTML
626            .replace("__FLOW_TITLE__", &escape_html(name))
627            .replace("__FLOW_DATA__", &json.replace("</", "<\\/")),
628    )?;
629    Ok(vec![json_path, html_path])
630}

632fn safe_filename(name: &str) -> String {
633    let filename: String = name
634        .chars()
635        .map(|character| {
636            if character.is_ascii_alphanumeric() || matches!(character, '-' | '_' | '.') {
637                character
638            } else {
639                '_'
640            }
641        })
642        .collect();
643    if filename.is_empty() {
644        "flow".to_string()
645    } else {
646        filename
647    }
648}

650fn escape_html(value: &str) -> String {
651    value
652        .replace('&', "&amp;")
653        .replace('<', "&lt;")
654        .replace('>', "&gt;")
655        .replace('"', "&quot;")
656        .replace('\'', "&#39;")
657}
