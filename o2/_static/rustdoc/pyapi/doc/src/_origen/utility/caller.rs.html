10#[derive(Debug)]
11pub struct FrameInfo {
12    filename: String,
13    lineno: usize,
14    function: String,
15    code_context: Option<Vec<String>>,
16    index: Option<usize>,
17}
18
19impl FrameInfo {
20    /// Turns the frame into an AST meta object, consuming self in the process
21    pub fn to_meta(self) -> Meta {
22        Meta {
23            filename: Some(self.filename),
24            lineno: Some(self.lineno),
25        }
26    }
27}
28
29enum Filter<'a> {
30    None,
31    StartsWith(&'a str),
32    Contains(&'a str),
33}
34
35/// Returns the final caller from Python, equivalent to caller::stack[0] but quicker to return if
36/// that's all you need from the stack.
37/// Returns None if an error occurred extracting the stack info.
38pub fn caller() -> Option<FrameInfo> {
39    let mut stack = match _get_stack(Some(1), Filter::None) {
40        Err(_e) => return None,
41        Ok(x) => x,
42    };
43    stack.pop()
44}
45
46/// Returns the last caller from Python application code, screening out any later calls through Origen
47/// core or plugin code
48pub fn app_caller() -> Option<FrameInfo> {
49    if let Some(app) = origen::app() {
50        if let Some(root) = app.root.to_str() {
51            let mut stack = match _get_stack(Some(1), Filter::StartsWith(root)) {
52                Err(_e) => return None,
53                Ok(x) => x,
54            };
55            return stack.pop();
56        }
57    }
58    None
59}

61/// Returns the last caller that was a test program flow (and possibly a pattern in future, hence the
62/// generic function name)
63pub fn src_caller() -> Option<FrameInfo> {
64    let file = origen::with_current_job(|job| {
65        if let Some(f) = job.files.last() {
66            Ok(Some(format!("{}", f.display())))
67        } else {
68            Ok(None)
69        }
70    })
71    .unwrap_or(None);
72    if let Some(f) = file {
73        caller_containing(&f)
74    } else {
75        None
76    }
77}

79/// Same as src_caller() but returns an AST metadata
80pub fn src_caller_meta() -> Option<Meta> {
81    if STATUS.is_debug_enabled() {
82        let c = src_caller();
83        let m = match c {
84            Some(m) => Some(m.to_meta()),
85            None => None,
86        };
87        m
88    } else {
89        None
90    }
91}

93/// Returns the last caller where the filename contains the given text
94pub fn caller_containing(text: &str) -> Option<FrameInfo> {
95    let mut stack = match _get_stack(Some(1), Filter::Contains(text)) {
96        Err(_e) => return None,
97        Ok(x) => x,
98    };
99    stack.pop()
100}

102/// Returns the full Python stack, including calls from app code, plugin code and Origen core
103/// Returns None if an error occurred extracting the stack info.
104pub fn stack() -> Option<Vec<FrameInfo>> {
105    match _get_stack(None, Filter::None) {
106        Err(_e) => {
107            //log_debug!("{:?}", e);
108            //let gil = Python::acquire_gil();
109            //let py = gil.python();
110            //e.print(py);
111            None
112        }
113        Ok(x) => Some(x),
114    }
115}

117fn _get_stack(max_depth: Option<usize>, filter: Filter) -> Result<Vec<FrameInfo>, PyErr> {
118    Python::with_gil(|py| {
119        let inspect = PyModule::import(py, "inspect")?;
120        let stack: Vec<Vec<&PyAny>> = inspect.getattr("stack")?.call0()?.extract()?;
121        let mut frames: Vec<FrameInfo> = vec![];
122        for f in stack {
123            let filename: String = f[1].extract()?;
124            let include = match filter {
125                Filter::None => true,
126                Filter::StartsWith(s) => filename.starts_with(s),
127                Filter::Contains(s) => filename.contains(s),
128            };
129            if include {
130                frames.push(FrameInfo {
131                    filename: filename,
132                    lineno: f[2].extract()?,
133                    function: f[3].extract()?,
134                    code_context: f[4].extract()?,
135                    index: f[5].extract()?,
136                });
137
138                if let Some(x) = max_depth {
139                    if x == frames.len() {
140                        break;
141                    }
142                }
143            }
144        }
145        Ok(frames)
146    })
