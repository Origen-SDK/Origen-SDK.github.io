39    /// Returns the application version, read from config/version.toml
40    pub fn version(&self) -> Result<Version> {
41        let version_file = self.root.join("pyproject.toml");
42        log_trace!("Reading app version");
43        if !version_file.exists() {
44            bail!("File does not exist '{}'", version_file.display());
45        }
46        let content = match fs::read_to_string(&version_file) {
47            Ok(x) => x,
48            Err(e) => bail!("There was a problem reading the app version file: {}", e),
49        };
50        lazy_static! {
51            static ref VERSION_LINE: Regex =
52                Regex::new(r#"(?m)^\s*version\s*=\s*['"](.*)['"]"#).unwrap();
53        }
54        if VERSION_LINE.is_match(&content) {
55            let captures = VERSION_LINE.captures(&content).unwrap();
56            Ok(Version::new_pep440(&captures[1])?)
57        } else {
58            bail!(
59                "Failed to read a version from file '{}'",
60                version_file.display()
61            )
62        }
63    }

69    /// Sets the application version by writing it out to config/version.toml
70    /// The normal way to do this is to call app.version(), bump the returned version object
71    /// as required, then return it back to this function.
72    /// See the [semver API](https://docs.rs/semver).
73    pub fn set_version(&self, version: &Version) -> Result<()> {
74        log_info!(
75            "Updating version file: '{}'",
76            self.version_file().into_os_string().into_string()?
77        );
78        let mut v = Version::from_pyproject_with_toml_handle(self.version_file())?;
79        v.set_new_version(version.clone())?;
80        v.write()?;
81        Ok(())
82    }

116    /// Return an RevisionControl, containing a driver, based on the app's config
117    pub fn rc(&self) -> Result<RevisionControl> {
118        self.with_config(|cfg| match cfg.revision_control.as_ref() {
119            Some(rc) => {
120                let mut resolved = rc.clone();
121                if let Some(local) = resolved.get("local") {
122                    let path = PathBuf::from(local);
123                    if path.is_relative() {
124                        resolved.insert(
125                            "local".to_string(),
126                            self.root.join(path).to_string_lossy().to_string(),
127                        );
128                    }
129                }
130                RevisionControl::from_config(&resolved)
131            }
132            None => bail!("No app RC was given. Cannot create RC driver"),
133        })
134    }

330    /// Resolves a directory/file path relative to the application's root.
331    /// Accepts an optional 'user_val' and a default. The resulting directory will be resolved from:
332    /// 1. If a user value is given, and its absolute, this is the final path.
333    /// 2. If a user value is given but its not absolute, then the final path is the user path relative to the application root.
334    /// 3. If no user value is given, the final path is the default path relative to the root.
335    /// Notes:
336    ///   A default is required, but an empty default will point to the application root.
337    ///   The default is assumed to be relative. Absolute defaults are not supported.
338    pub fn resolve_path(&self, user_val: Option<&String>, default: &str) -> PathBuf {
339        let offset;
340        if let Some(user_str) = user_val {
341            if Path::new(&user_str).is_absolute() {
342                return PathBuf::from(user_str);
343            } else {
344                offset = user_str.to_string();
345            }
346        } else {
347            offset = default.to_string();
348        }
349        let mut dir = self.root.clone();
350        dir.push(offset);
351        dir
352    }

385pub struct ProductionStatus {
386    checks: IndexMap<String, (bool, Option<String>, Option<String>)>,
387    passed: bool,
388}
389
390impl Default for ProductionStatus {
391    fn default() -> Self {
392        Self {
393            checks: IndexMap::new(),
394            passed: true,
395        }
396    }

399impl ProductionStatus {
400    pub fn push_check(
401        &mut self,
402        check: &str,
403        desc: Option<String>,
404        result: bool,
405        result_text: Option<String>,
406    ) {
407        self.checks
408            .insert(check.to_string(), (result, desc, result_text));
409        if !result {
410            self.passed = false;
411        }
412    }

418            clean,

427            passed,
428            txt,
429        );

440    pub fn summarize_and_exit(&self) {
441        if self.passed() {
442            exit_pass!();
443        } else {
444            for (n, check) in self.checks.iter() {
445                if check.0 {
446                    display_greenln!("{}... PASSED!", n);
447                } else {
448                    if let Some(m) = check.2.as_ref() {
449                        display_redln!("{}... FAILED with message: {}", n, m);
450                    } else {
451                        display_redln!("{}... FAILED", n);
452                    }
453                }
454            }
455            exit_fail!();
456        }
457    }
