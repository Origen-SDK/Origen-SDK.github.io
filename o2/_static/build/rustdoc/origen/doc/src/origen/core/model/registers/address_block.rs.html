1use super::{AccessType, MemoryMap};
2use crate::core::model::Model;
3use crate::Dut;
4use crate::Result as OrigenResult;
5use indexmap::map::IndexMap;
6use std::sync::MutexGuard;
7
8#[derive(Debug)]
9/// Represents a single, contiguous block of memory in a memory map.
10pub struct AddressBlock {
11    pub id: usize,
12    pub memory_map_id: usize,
13    pub name: String,
14    /// The starting address of the address block expressed in address_unit_bits
15    /// from the parent memory map.
16    pub offset: u128,
17    /// The number of addressable units in the address block.
18    pub range: u64,
19    /// The maximum number of bits that can be accessed by a transaction into this
20    /// address block.
21    pub width: Option<u64>,
22    pub access: AccessType,
23    pub registers: IndexMap<String, usize>,
24    pub register_files: IndexMap<String, usize>,
25}
26
27impl Default for AddressBlock {
28    fn default() -> AddressBlock {
29        AddressBlock {
30            id: 0,
31            memory_map_id: 0,
32            name: "default".to_string(),
33            offset: 0,
34            range: 0,
35            width: None,
36            access: AccessType::RW,
37            registers: IndexMap::new(),
38            register_files: IndexMap::new(),
39        }
40    }
41}
42
43impl AddressBlock {
44    /// Returns an immutable reference to the parent model
45    pub fn model<'a>(&self, dut: &'a MutexGuard<Dut>) -> OrigenResult<&'a Model> {
46        self.memory_map(dut)?.model(dut)
47    }
48
49    /// Returns an immutable reference to the parent memory map
50    pub fn memory_map<'a>(&self, dut: &'a MutexGuard<Dut>) -> OrigenResult<&'a MemoryMap> {
51        dut.get_memory_map(self.memory_map_id)
52    }
53
54    /// Returns the address_unit_bits size from the parent memory map
55    pub fn address_unit_bits(&self, dut: &MutexGuard<Dut>) -> OrigenResult<u32> {
56        Ok(self.memory_map(dut)?.address_unit_bits)
57    }
58
59    /// Returns the fully-resolved address taking into account all base addresses defined by the parent hierachy.
60    /// The returned address is with an address_unit_bits size of 1.
61    pub fn bit_address(&self, dut: &MutexGuard<Dut>) -> OrigenResult<u128> {
62        let base = self.model(dut)?.bit_address(dut)?;
63        Ok(base + (self.offset * self.address_unit_bits(dut)? as u128))
64    }
65
66    /// Returns a path to this address_block like "dut.my_block.my_map.my_address_block", but the map and address block portions
67    /// will be inhibited when they are 'default'. This is to keep map and address block concerns out of the view of users who
68    /// don't use them and simply define regs at the top-level of the block.
69    pub fn friendly_path(&self, dut: &MutexGuard<Dut>) -> OrigenResult<String> {
70        let path = self.memory_map(dut)?.friendly_path(dut)?;
71        if self.name == "default" {
72            Ok(path)
73        } else {
74            Ok(format!("{}.{}", path, self.name))
75        }
76    }
77
78    /// Get the ID from the given register name
79    pub fn get_register_id(&self, name: &str) -> OrigenResult<usize> {
80        match self.registers.get(name) {
81            Some(x) => Ok(*x),
82            None => {
83                bail!(
84                    "The address block '{}' does not have a register named '{}'",
85                    self.name,
86                    name
87                )
88            }
89        }
90    }
91
92    pub fn console_display(&self, dut: &MutexGuard<Dut>) -> OrigenResult<String> {
93        let (mut output, offset) = self.model(dut)?.console_header(dut);
94        output += &(" ".repeat(offset));
95        output += &format!("└── memory_maps['{}']\n", self.memory_map(dut)?.name);
96        let mut leader = " ".repeat(offset + 5);
97        output += &format!("{}└── address_blocks['{}']\n", leader, self.name);
98        leader += "     ";
99        let num = self.register_files.keys().len();
100        if num > 0 {
101            output += &format!("{}├── register_files\n", leader);
102            let leader = format!("{}|    ", leader);
103            for (i, key) in self.register_files.keys().enumerate() {
104                if i != num - 1 {
105                    output += &format!("{}├── {}\n", leader, key);
106                } else {
107                    output += &format!("{}└── {}\n", leader, key);
108                }
109            }
110        } else {
111            output += &format!("{}├── register_files []\n", leader);
112        }
113        let num = self.registers.keys().len();
114        if num > 0 {
115            output += &format!("{}└── registers\n", leader);
116            let leader = format!("{}     ", leader);
117            for (i, key) in self.registers.keys().enumerate() {
118                if i != num - 1 {
119                    output += &format!("{}├── {}\n", leader, key);
120                } else {
121                    output += &format!("{}└── {}\n", leader, key);
122                }
123            }
124        } else {
125            output += &format!("{}├── registers []\n", leader);
126        }
127        Ok(output)
128    }
129}
