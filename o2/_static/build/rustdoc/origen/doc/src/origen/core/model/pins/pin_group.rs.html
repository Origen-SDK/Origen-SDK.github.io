7// We'll maintain both the pin_names which the group was built with, but we'll also maintain the list
8// of physical names. Even though we can resolve this later, most operations wil
9#[derive(Debug, Clone)]
10pub struct PinGroup {
11    pub model_id: usize,
12    pub id: usize,
13    pub name: String,
14    pub endianness: Endianness,
15    pub pin_ids: Vec<usize>,
16}
17
18impl PinGroup {
19    pub fn new(
20        model_id: usize,
21        id: usize,
22        name: &str,
23        pin_ids: Vec<usize>,
24        endianness: Option<Endianness>,
25    ) -> PinGroup {
26        return PinGroup {
27            model_id: model_id,
28            id: id,
29            name: name.to_string(),
30            pin_ids: match endianness {
31                Some(e) => match e {
32                    Endianness::LittleEndian => pin_ids,
33                    Endianness::BigEndian => {
34                        let mut _pins = pin_ids.clone();
35                        _pins.reverse();
36                        _pins
37                    }
38                },
39                None => pin_ids,
40            },
41            endianness: match endianness {
42                Some(e) => e,
43                None => Endianness::LittleEndian,
44            },
45        };
46    }
47
48    pub fn len(&self) -> usize {
49        return self.pin_ids.len();
50    }
51
52    pub fn is_little_endian(&self) -> bool {
53        match self.endianness {
54            Endianness::LittleEndian => true,
55            Endianness::BigEndian => false,
56        }
57    }
58
59    pub fn is_big_endian(&self) -> bool {
60        return !self.is_little_endian();
61    }
62
63    pub fn to_identifier(&self) -> (String, usize) {
64        (self.name.clone(), self.model_id)
65    }
66
67    pub fn pin_names(&self, dut: &Dut) -> Result<Vec<String>> {
68        let pc = super::super::pins::PinCollection::from_pin_group(&dut, self)?;
69        Ok(pc.pin_names())
70    }
71
72    pub fn update(&self, dut: &Dut, trans: &Transaction) -> Result<()> {
73        let pc = super::super::pins::PinCollection::from_pin_group(&dut, self)?;
74        pc.set_from_transaction(trans)?;
75        Ok(())
76    }
77
78    pub fn overlay(&self, overlay: &mut Overlay) -> Result<()> {
79        overlay.pin_ids = Some(self.pin_ids.clone());
80        crate::TEST.push(overlay.to_node());
81        Ok(())
82    }
83
84    pub fn capture(&self, capture: &mut Capture) -> Result<()> {
85        capture.pin_ids = Some(self.pin_ids.clone());
86        crate::TEST.push(capture.to_node());
87        Ok(())
88    }
89
90    pub fn get_actions(&self, dut: &Dut) -> Result<Vec<PinAction>> {
91        let pc = super::super::pins::PinCollection::from_pin_group(&dut, self)?;
92        Ok(pc.get_actions())
93    }
94
95    pub fn get_reset_actions(&self, dut: &Dut) -> Result<Vec<PinAction>> {
96        let pc = super::super::pins::PinCollection::from_pin_group(&dut, self)?;
97        Ok(pc.get_reset_actions())
98    }
99
100    pub fn reset(&self, dut: &Dut) -> Result<()> {
101        let pc = super::super::pins::PinCollection::from_pin_group(&dut, self)?;
102        pc.reset();
103        Ok(())
104    }
105
106    pub fn set_action(&self, dut: &Dut, action: &PinAction) -> Result<()> {
107        let pc = super::super::pins::PinCollection::from_pin_group(&dut, &self)?;
108        pc.set_action(&action.to_string());
109        Ok(())
110    }
111
112    pub fn set_actions(&self, dut: &Dut, actions: &Vec<PinAction>) -> Result<()> {
113        let pc = super::super::pins::PinCollection::from_pin_group(&dut, &self)?;
114        let mut v = vec![];
115        actions.iter().for_each(|a| v.push(a.to_string()));
116        pc.set_actions(&v)?;
117        Ok(())
118    }
119
120    pub fn slice(&self, start_idx: usize, stop_idx: usize, step_size: usize) -> Result<PinStore> {
121        let mut sliced_names: Vec<usize> = vec![];
122        for i in (start_idx..=stop_idx).step_by(step_size) {
123            if i >= self.pin_ids.len() {
124                bail!(
125                    "Index {} exceeds available pins in collection! (length: {})",
126                    i,
127                    self.pin_ids.len()
128                );
129            }
130            let p = self.pin_ids[i].clone();
131            sliced_names.push(p);
132        }
133        Ok(PinStore::new(sliced_names, Some(self.endianness)))
134    }
