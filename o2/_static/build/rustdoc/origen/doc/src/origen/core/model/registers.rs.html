24#[derive(Debug, PartialEq, Copy, Clone)]
25pub enum AccessType {
26    RO,            // Read-Only
27    RW,            // Read-Write
28    RC,            // Read-only, Clear-on-read
29    RS,            // Set-on-read (all bits become '1' on read)
30    WRC,           // Writable, clear-on-read
31    WRS,           // Writable, Sets-on-read
32    WC,            // Clear-on-write
33    WS,            // Set-on-write
34    WSRC,          // Set-on-write, clear-on-read
35    WCRS,          // Clear-on-write, set-on-read
36    W1C,           // Write '1' to clear bits
37    W1S,           // Write '1' to set bits
38    W1T,           // Write '1' to toggle bits
39    W0C,           // Write '0' to clear bits
40    W0S,           // Write '0' to set bits
41    W0T,           // Write '0' to toggle bits
42    W1SRC,         // Write '1' to set and clear-on-read
43    W1CRS,         // Write '1' to clear and set-on-read
44    W0SRC,         // Write '0' to set and clear-on-read
45    W0CRS,         // Write '0' to clear and set-on-read
46    WO,            // Write-only
47    WOC,           // When written sets the field to '0'. Read undeterministic
48    WORZ,          // Write-only, Reads zero
49    WOS,           // When written sets all bits to '1'. Read undeterministic
50    W1,            // Write-once. Next time onwards, write is ignored. Read returns the value
51    WO1,           // Write-once. Next time onwards, write is ignored. Read is undeterministic
52    DC,            // RW but no check
53    ROWZ,          // Read-only, value is cleared on read
54    Unimplemented, // Means an unimplemented bit in the Origen model
55}

57impl std::str::FromStr for AccessType {
58    type Err = String;
59
60    fn from_str(s: &str) -> Result<Self, Self::Err> {
61        match s {
62            "RO" | "ro" => Ok(AccessType::RO),
63            "RW" | "rw" => Ok(AccessType::RW),
64            "RC" | "rc" => Ok(AccessType::RC),
65            "RS" | "rs" => Ok(AccessType::RS),
66            "WRC" | "wrc" => Ok(AccessType::WRC),
67            "WRS" | "wrs" => Ok(AccessType::WRS),
68            "WC" | "wc" => Ok(AccessType::WC),
69            "WS" | "ws" => Ok(AccessType::WS),
70            "WSRC" | "wsrc" => Ok(AccessType::WSRC),
71            "WCRS" | "wcrs" => Ok(AccessType::WCRS),
72            "W1C" | "w1c" => Ok(AccessType::W1C),
73            "W1S" | "w1s" => Ok(AccessType::W1S),
74            "W1T" | "w1t" => Ok(AccessType::W1T),
75            "W0C" | "w0c" => Ok(AccessType::W0C),
76            "W0S" | "w0s" => Ok(AccessType::W0S),
77            "W0T" | "w0t" => Ok(AccessType::W0T),
78            "W1SRC" | "w1src" => Ok(AccessType::W1SRC),
79            "W1CRS" | "w1crs" => Ok(AccessType::W1CRS),
80            "W0SRC" | "w0src" => Ok(AccessType::W0SRC),
81            "W0CRS" | "w0crs" => Ok(AccessType::W0CRS),
82            "WO" | "wo" => Ok(AccessType::WO),
83            "WOC" | "woc" => Ok(AccessType::WOC),
84            "WORZ" | "worz" => Ok(AccessType::WORZ),
85            "WOS" | "wos" => Ok(AccessType::WOS),
86            "W1" | "w1" => Ok(AccessType::W1),
87            "WO1" | "wo1" => Ok(AccessType::WO1),
88            "DC" | "dc" => Ok(AccessType::DC),
89            "ROWZ" | "rowz" => Ok(AccessType::ROWZ),
90            _ => Err(format!("'{}' is not a valid value for AccessType", s)),
91        }
92    }
93}

103impl AccessType {
104    pub fn is_readable(&self) -> bool {
105        *self != AccessType::WO
106    }
107
108    pub fn is_writeable(&self) -> bool {
109        *self != AccessType::RO && *self != AccessType::Unimplemented
110    }
111
112    pub fn is_writable(&self) -> bool {
113        self.is_writeable()
114    }
115
116    pub fn is_unimplemented(&self) -> bool {
117        *self == AccessType::Unimplemented
118    }
119}

122pub enum BitOrder {

125}
126
127impl std::str::FromStr for BitOrder {
128    type Err = String;
129
130    fn from_str(s: &str) -> Result<Self, Self::Err> {
131        match s {
132            "MSB0" | "msb0" => Ok(BitOrder::MSB0),
133            "LSB0" | "lsb0" => Ok(BitOrder::LSB0),
134            _ => Err(format!("'{}' is not a valid value for BitOrder", s)),
135        }
136    }
137}
