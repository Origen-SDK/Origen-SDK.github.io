6#[derive(Debug, PartialEq, Clone, Eq, Hash, Serialize)]
7pub enum SupportedTester {
8    /// Generally, the absence of an optional SupportedTester value means all testers, but
9    /// this can also be used to indicate that whenever a SupportedTester value is required
10    ALL,
11    /// Indicates support for all versions of SMT on the V93K
12    V93K,
13    /// Indicates support for V93K SMT7 only, i.e. not SMT8
14    V93KSMT7,
15    /// Indicates support for V93K SMT8 only, i.e. not SMT7
16    V93KSMT8,
17    /// Indicates support for all IGXL-based testers
18    IGXL,
19    J750,
20    ULTRAFLEX,
21    SIMULATOR,
22    // These two are only available in an Origen workspace
23    DUMMYRENDERER,
24    DUMMYRENDERERWITHINTERCEPTORS,
25    // Used to identify an app-defined tester (in Python)
26    CUSTOM(String),
27}
28
29impl SupportedTester {
30    /// Returns the names of all available testers
31    pub fn all_names() -> Vec<String> {
32        let mut s = vec![
33            "ALL",
34            "V93K",
35            "V93KSMT7",
36            "V93KSMT8",
37            "IGXL",
38            "J750",
39            "ULTRAFLEX",
40            "SIMULATOR",
41        ];
42        if crate::STATUS.is_origen_present {
43            s.push("DUMMYRENDERER");
44            s.push("DUMMYRENDERERWITHINTERCEPTORS");
45        }
46        let mut s: Vec<String> = s.iter().map(|n| n.to_string()).collect();
47        for id in crate::STATUS.custom_tester_ids() {
48            s.push(format!("CUSTOM::{}", id));
49        }
50        s
51    }
52
53    pub fn prog_gen_supported_tester(&self) -> OrigenResult<ProgGenSupportedTester> {
54        match self {
55            SupportedTester::ALL => Ok(ProgGenSupportedTester::ALL),
56            SupportedTester::V93K => Ok(ProgGenSupportedTester::V93K),
57            SupportedTester::V93KSMT7 => Ok(ProgGenSupportedTester::V93KSMT7),
58            SupportedTester::V93KSMT8 => Ok(ProgGenSupportedTester::V93KSMT8),
59            SupportedTester::IGXL => Ok(ProgGenSupportedTester::IGXL),
60            SupportedTester::J750 => Ok(ProgGenSupportedTester::J750),
61            SupportedTester::ULTRAFLEX => Ok(ProgGenSupportedTester::ULTRAFLEX),
62            SupportedTester::SIMULATOR => Ok(ProgGenSupportedTester::SIMULATOR),
63            _ => bail!("No equivalent prog gen tester exists for {:?}", self),
64        }
65    }
66
67    pub fn new(name: &str) -> OrigenResult<Self> {
68        match SupportedTester::from_str(name) {
69            Ok(n) => Ok(n),
70            Err(msg) => bail!("{}", msg),
71        }
72    }
73
74    /// Returns true if the given tester is equal to self or if the given tester is a
75    /// derivative of self (see is_a_derivative_of()).
76    pub fn is_compatible_with(&self, tester: &SupportedTester) -> bool {
77        self == tester || tester.is_a_derivative_of(self)
78    }
79
80    /// Returns true if self is a derivative of the given tester. For example if the given
81    /// tester is IGXL, then both the J750 and the ULTRAFLEX would be considered derivatives
82    /// of it.
83    /// Note that true is only returned for derivatives, if the given tester is the same as
84    /// self then false will be returned.
85    /// Use is_compatible_with() if you also want to consider an exact match as true.
86    pub fn is_a_derivative_of(&self, tester: &SupportedTester) -> bool {
87        match self {
88            SupportedTester::ALL => tester != &SupportedTester::ALL,
89            SupportedTester::IGXL => {
90                tester == &SupportedTester::J750 || tester == &SupportedTester::ULTRAFLEX
91            }
92            SupportedTester::V93K => {
93                tester == &SupportedTester::V93KSMT7 || tester == &SupportedTester::V93KSMT8
94            }
95            _ => false,
96        }
97    }
98}

100impl FromStr for SupportedTester {
101    type Err = String;
102
103    fn from_str(s: &str) -> Result<Self, Self::Err> {
104        let mut second: Option<String> = None;
105        let kind = match s.contains("::") {
106            true => {
107                let fields: Vec<&str> = s.split("::").collect();
108                if fields.len() > 2 {
109                    return Err(error_msg(&s));
110                }
111                second = Some(fields[1].to_string());
112                fields[0]
113            }
114            false => s,
115        };
116
117        // Accept any case and with or without underscores
118        let kind = kind.to_uppercase().replace("_", "");
119        match kind.trim() {
120            "ALL" | "ANY" => Ok(SupportedTester::ALL),
121            "V93K" => Ok(SupportedTester::V93K),
122            "V93KSMT7" => Ok(SupportedTester::V93KSMT7),
123            "V93KSMT8" => Ok(SupportedTester::V93KSMT8),
124            "IGXL" => Ok(SupportedTester::IGXL),
125            "J750" => Ok(SupportedTester::J750),
126            "ULTRAFLEX" | "UFLEX" => Ok(SupportedTester::ULTRAFLEX),
127            "SIMULATOR" => Ok(SupportedTester::SIMULATOR),
128            "DUMMYRENDERER" => Ok(SupportedTester::DUMMYRENDERER),
129            "DUMMYRENDERERWITHINTERCEPTORS" => Ok(SupportedTester::DUMMYRENDERERWITHINTERCEPTORS),
130            "CUSTOM" => {
131                if let Some(n) = second {
132                    if crate::STATUS.custom_tester_ids().contains(&n) {
133                        Ok(SupportedTester::CUSTOM(n))
134                    } else {
135                        Err(error_msg(&s))
136                    }
137                } else {
138                    Err(error_msg(&s))
139                }
140            }
141            _ => Err(error_msg(&s)),
142        }
143    }
144}

152fn error_msg(val: &str) -> String {
153    format!(
154        "'{}' is not a valid tester type, the available testers are: {}",
155        val,
156        SupportedTester::all_names().join(", ")
157    )
158}

165    #[test]
166    fn verify_custom_testers_work_as_hash_keys() {
167        let mut h: HashMap<SupportedTester, usize> = HashMap::new();
168        let t1 = SupportedTester::J750;
169        let t2 = SupportedTester::CUSTOM("t2".to_string());
170        let t3 = SupportedTester::CUSTOM("t3".to_string());
171
172        h.insert(t1.clone(), 1);
173        h.insert(t2.clone(), 2);
174        h.insert(t3.clone(), 3);
175
176        assert_eq!(h[&t1], 1);
177        assert_eq!(h[&t2], 2);
178        assert_eq!(h[&t3], 3);
179    }
