26impl CurrentState {
27    pub fn build(root: &PathBuf) -> Self {
28        let file = root.join(".origen").join("application.toml");
29        let mut s = config::Config::builder()
30            .set_default("target", None::<Vec<String>>)
31            .unwrap();
32        if file.exists() {
33            s = s.add_source(File::with_name(&format!("{}", file.display())));
34        }
35        let cb = exit_on_bad_config!(s.build());
36        let slf: Self = exit_on_bad_config!(cb.try_deserialize());
37        slf
38    }
39
40    pub fn apply_to(&mut self, config: &mut Config) {
41        if let Some(t) = self.target.as_ref() {
42            config.target = Some(t.to_owned())
43        } else {
44            if let Some(t) = &config.target {
45                let clean_defaults = target::set_at_root(
46                    t.iter().map(|s| s.as_str()).collect(),
47                    config.root.as_ref().unwrap(),
48                );
49                self.target = Some(clean_defaults);
50            }
51        }
52    }
53
54    pub fn build_and_apply(config: &mut Config) {
55        if use_app_config!() {
56            let mut slf = Self::build(config.root.as_ref().unwrap());
57            slf.apply_to(config);
58        }
59    }
60}

113    /// Builds a new config from all application.toml files found at the given app root
114    pub fn build(root: &Path, default_only: bool) -> Config {
115        log_trace!("Building app config");
116        let mut s = config::Config::builder()
117            .set_default("target", None::<Vec<String>>)
118            .unwrap()
119            .set_default("mode", "development".to_string())
120            .unwrap()
121            .set_default("revision_control", None::<HashMap<String, String>>)
122            .unwrap()
123            .set_default("unit_tester", None::<HashMap<String, String>>)
124            .unwrap()
125            .set_default("publisher", None::<HashMap<String, String>>)
126            .unwrap()
127            .set_default("linter", None::<HashMap<String, String>>)
128            .unwrap()
129            .set_default("release_scribe", None::<HashMap<String, String>>)
130            .unwrap()
131            .set_default("release", None::<HashMap<String, String>>)
132            .unwrap()
133            .set_default("app_session_root", None::<String>)
134            .unwrap()
135            .set_default("commands", None::<Vec<String>>)
136            .unwrap();
137
138        let mut files: Vec<PathBuf> = Vec::new();
139        if let Some(paths) = std::env::var_os(APP_CONFIG_PATHS) {
140            log_trace!("Found custom config paths: {:?}", paths);
141            for path in std::env::split_paths(&paths) {
142                log_trace!("Looking for Origen app config file at '{}'", path.display());
143                if path.is_file() {
144                    if let Some(ext) = path.extension() {
145                        if ext == "toml" {
146                            files.push(path);
147                        } else {
148                            log_error!(
149                                "Expected file {} to have extension '.toml'. Found '{}'",
150                                path.display(),
151                                ext.to_string_lossy()
152                            )
153                        }
154                    } else {
155                        // accept a file without an extension. will be interpreted as a .toml
156                        files.push(path);
157                    }
158                } else if path.is_dir() {
159                    let f = path.join("application.toml");
160                    if f.exists() {
161                        files.push(f);
162                    }
163                } else {
164                    log_error!(
165                        "Config path {} either does not exists or is not accessible",
166                        path.display()
167                    );
168                    exit(1);
169                }
170            }
171        }
172
173        if use_app_config!() {
174            let file = root.join("config").join("application.toml");
175            if file.exists() {
176                files.push(file);
177            }
178        } else {
179            // Bypass Origen's default configuration lookup - use only the enumerated configs
180            log_trace!("Bypassing Origen's App Config Lookup");
181        }
182
183        for f in files.iter().rev() {
184            log_trace!("Loading Origen config file from '{}'", f.display());
185            s = s.add_source(File::with_name(&format!("{}", f.display())));
186        }
187        s = s.add_source(
188            Environment::with_prefix("origen_app")
189                .list_separator(",")
190                .with_list_parse_key("target")
191                .with_list_parse_key("commands")
192                .try_parsing(true),
193        );
194
195        let cb = exit_on_bad_config!(s.build());
196        let mut c: Self = exit_on_bad_config!(cb.try_deserialize());
197        c.root = Some(root.to_path_buf());
198        // TODO
199        // if let Some(l) = loc {
200        //     c.website_release_location = Some(Location::new(&l));
201        // }
202        log_trace!("Completed building app config");
203        c.validate_options();
204        if !default_only {
205            CurrentState::build_and_apply(&mut c);
206        }
207
208        c
209    }

211    pub fn validate_options(&self) {
212        log_trace!("Validating available options...");
213
214        if let Some(targets) = self.target.as_ref() {
215            log_trace!("\tValidating default target...");
216            for t in targets {
217                target::clean_name(t, "targets", true, self.root.as_ref().unwrap());
218            }
219            log_trace!("\tValidating default target!");
220        }
221
222        log_trace!("\tValidating publisher options...");
223        for unknown in self.validate_publisher_options() {
224            log_warning!("Unknown Publisher Option '{}'", unknown);
225        }
226        log_trace!("\tFinished validating publisher options");
227        log_trace!("Finished checking configs!");
228    }
229
230    pub fn validate_publisher_options(&self) -> Vec<String> {
231        let mut unknowns: Vec<String> = vec![];
232        if let Some(p) = &self.publisher {
233            for (opt, _) in p.iter() {
234                if !PUBLISHER_OPTIONS.contains(&opt.as_str()) {
235                    unknowns.push(opt.clone());
236                }
237            }
238        }
239        unknowns
240    }

242    pub fn cmd_paths(&self) -> Vec<PathBuf> {
243        let mut retn = vec![];
244        if let Some(cmds) = self.commands.as_ref() {
245            // Load in only the commands explicitly given
246            for cmds_toml in cmds {
247                let ct = self.root.as_ref().unwrap().join("config").join(cmds_toml);
248                if ct.exists() {
249                    retn.push(ct.to_owned());
250                } else {
251                    log_error!(
252                        "Can not locate app commands file '{}'",
253                        scrub_path!(ct).display()
254                    )
255                }
256            }
257        } else {
258            // Load in any commands from:
259            // 1) app_root/commands.toml
260            // 2) app_root/commands/*/**.toml
261            let commands_toml = self
262                .root
263                .as_ref()
264                .unwrap()
265                .join("config")
266                .join("commands.toml");
267            // println!("commands toml: {}", commands_toml.display());
268            if commands_toml.exists() {
269                retn.push(commands_toml);
270            }
271            let mut commands_dir = self.root.as_ref().unwrap().join("config/commands");
272            if commands_dir.exists() {
273                commands_dir = commands_dir.join("**/*.toml");
274                for entry in glob(commands_dir.to_str().unwrap()).unwrap() {
275                    match entry {
276                        Ok(e) => retn.push(e),
277                        Err(e) => log_error!("Error processing commands toml: {}", e),
278                    }
279                }
280            }
281        }
282        retn
283    }
