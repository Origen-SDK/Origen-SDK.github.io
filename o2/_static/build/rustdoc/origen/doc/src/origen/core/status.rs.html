19#[derive(Debug, Copy, Clone, PartialEq)]
20pub enum Operation {
21    None,
22    Convert,
23    Generate,
24    GeneratePattern,
25    GenerateFlow,
26    Compile,
27    Interactive,
28    Web,
29    Credentials,
30    App,
31    AppCommand,
32}
33
34impl FromStr for Operation {
35    type Err = String;
36
37    fn from_str(s: &str) -> Result<Self, Self::Err> {
38        // Accept any case
39        let s = s.to_lowercase();
40        match s.trim() {
41            "none" => Ok(Operation::None),
42            "convert" => Ok(Operation::Convert),
43            "generate" => Ok(Operation::Generate),
44            "generatepattern" => Ok(Operation::GeneratePattern),
45            "generateflow" => Ok(Operation::GenerateFlow),
46            "compile" => Ok(Operation::Compile),
47            "interactive" => Ok(Operation::Interactive),
48            "web" => Ok(Operation::Web),
49            "credentials" => Ok(Operation::Credentials),
50            "app" => Ok(Operation::App),
51            "appcommand" => Ok(Operation::AppCommand),
52            _ => Err(format!("Unknown Operation: '{}'", &s)),
53        }
54    }
55}

57#[derive(Debug, Display)]
58pub enum DependencySrc {
59    // Dependencies resolve from...
60    App(PathBuf),        // the application
61    Workspace(PathBuf),  // current directory tree (in workspace)
62    UserGlobal(PathBuf), // explicitly given by user, no workspace
63    Global(PathBuf),     // the origen CLI installation directory, no workspace
64    NoneFound, // None available. Use whatever is available in the same install environment as Origen itself
65}
66
67impl DependencySrc {
68    pub fn src_available(&self) -> bool {
69        match self {
70            Self::NoneFound => false,
71            _ => true,
72        }
73    }
74
75    pub fn src_file(&self) -> Option<&PathBuf> {
76        match self {
77            Self::App(path)
78            | Self::Workspace(path)
79            | Self::UserGlobal(path)
80            | Self::Global(path) => Some(path),
81            Self::NoneFound => None,
82        }
83    }
84}

86impl<S> TryFrom<(S, Option<PathBuf>)> for DependencySrc
87where
88    S: AsRef<str>,
89{
90    type Error = crate::Error;
91
92    fn try_from(value: (S, Option<PathBuf>)) -> Result<Self, Self::Error> {
93        macro_rules! gen_case {
94            ($t: ident) => {
95                if let Some(path) = value.1 {
96                    Self::$t(path)
97                } else {
98                    bail!(concat!(
99                        "A path is required with dependency src type '",
100                        stringify!($t),
101                        "'"
102                    ));
103                }
104            };
105        }
106        Ok(match value.0.as_ref() {
107            "App" => gen_case!(App),
108            "Workspace" => gen_case!(Workspace),
109            "UserGlobal" => gen_case!(UserGlobal),
110            "Global" => gen_case!(Global),
111            "NoneFound" => Self::NoneFound,
112            _ => bail!(
113                "Cannot convert value '{}' to dependency src type",
114                value.0.as_ref()
115            ),
116        })
117    }
118}

184impl Default for Status {
185    fn default() -> Status {
186        log_trace!("Building STATUS");
187        log_trace!(
188            "Current exe is {}",
189            std::env::current_exe().unwrap().display()
190        );
191        let mut dev_mode_origen_root: Option<PathBuf> = None;
192        let mut origen_dev_mode = false;
193        let (app_present, app_root) = search_for_from_pwd(vec!["config", "origen.toml"], true);
194        let (origen_present, origen_wksp_root) =
195            search_for_from_pwd(vec![".origen_dev_workspace"], false);
196
197        // If a standalone app is present check if it contains a reference to a local Origen
198        if app_present && !origen_present {
199            let pyproject = app_root.join("pyproject.toml");
200            match std::fs::read_to_string(&pyproject) {
201                Ok(contents) => {
202                    // (?m) is how you set the multiline flag
203                    let regex = Regex::new(
204                        r#"(?m)^\s*origen\s*=\s*\{\s*path\s*=\s*['"](?P<path>[^'"]+)['"]"#,
205                    )
206                    .unwrap();
207                    if let Some(captures) = regex.captures(&contents) {
208                        let path = PathBuf::from(captures.name("path").unwrap().as_str());
209                        let path = match path.is_relative() {
210                            true => app_root.join(&path),
211                            false => path,
212                        };
213                        let (found, path) = search_for(vec![".origen_dev_workspace"], false, &path);
214                        if found {
215                            dev_mode_origen_root = Some(path);
216                            origen_dev_mode = true;
217                        }
218                    }
219                }
220                Err(e) => log_error!("{}", e),
221            }
222        }
223
224        let version = match Version::new_semver(built_info::PKG_VERSION) {
225            Ok(v) => v,
226            Err(_e) => Version::default(),
227        };
228        let s = Status {
229            is_app_present: app_present,
230            app: match app_present {
231                true => Some(Application::new(app_root.clone())),
232                false => None,
233            },
234            is_origen_present: origen_present,
235            origen_wksp_root: match dev_mode_origen_root {
236                Some(x) => x,
237                None => origen_wksp_root,
238            },
239            origen_version: version,
240            start_time: time::now(),
241            home: get_home_dir(),
242            log_level: 1,
243            unhandled_error_count: RwLock::new(0),
244            output_dir: RwLock::new(None),
245            reference_dir: RwLock::new(None),
246            cli_location: RwLock::new(None),
247            cli_version: RwLock::new(None),
248            fe_pkg_loc: RwLock::new(None),
249            fe_exe_loc: RwLock::new(None),
250            origen_core_support_version: Version::new_semver({
251                let mut v: &str = "";
252                for d in built_info::DEPENDENCIES.iter() {
253                    if d.0 == "origen-core-support" {
254                        v = d.1;
255                        break;
256                    }
257                }
258                if v == "" {
259                    panic!("Could not determine origen-core-support version")
260                }
261                v
262            })
263            .unwrap(),
264            origen_metal_backend_version: Version::new_semver(&origen_metal::VERSION).unwrap(),
265            other_build_info: RwLock::new(HashMap::new()),
266            is_app_in_origen_dev_mode: origen_dev_mode,
267            in_origen_core_app: RwLock::new(false),
268            _custom_tester_ids: RwLock::new(vec![]),
269            testers_eq: RwLock::new(vec![]),
270            testers_neq: RwLock::new(vec![]),
271            unique_id: RwLock::new(0),
272            debug_enabled: RwLock::new(false),
273            _operation: RwLock::new(Operation::None),
274            dependency_src: RwLock::new(None),
275            // command: RwLock::new(None), // FEATURE Backend Current Command
276        };
277        log_trace!("Status built successfully");
278        s
279    }

521    /// This is the main method to get the current output directory, accounting for all
522    /// possible ways to set it, from current command, the app, default, etc.
523    /// If nothing has been set (only possible when running globally), then it will default
524    /// to the PWD.
525    /// It will ensure that the directory exists before returning it.
526    pub fn output_dir(&self) -> PathBuf {
527        let mut dir = self.output_dir.read().unwrap().to_owned();
528        // If it hasn't been set by the current command
529        if dir.is_none() {
530            if let Some(app) = &self.app {
531                dir = Some(app._output_directory());
532            } else {
533                dir = Some(env::current_dir().expect(
534                    "Can't read the current directory when trying to set the output directory",
535                ));
536            }
537        }
538        let dir = dir.unwrap();
539        if !dir.exists() {
540            std::fs::create_dir_all(&dir).expect(&format!(
541                "Couldn't create the output directory '{}'",
542                dir.display()
543            ));
544        }
545        dir
546    }

549    /// Optionally, the current working directory can be switched to the output dir before executing
550    /// the function and then restored at the end by setting change_to to True.
551    /// If this is called when Origen is executing outside of an application workspace then it will

553    pub fn with_output_dir<T, F>(&self, change_dir: bool, mut f: F) -> OrigenResult<T>

585        } else {
586            None
587        }

591    /// Optionally, the current working directory can be switched to the reference dir before executing
592    /// the function and then restored at the end by setting change_to to True.
593    /// If this is called when Origen is executing outside of an application workspace then it will
594    /// return None unless it has been previously set by calling set_reference_dir().
595    pub fn with_reference_dir<T, F>(&self, change_dir: bool, mut f: F) -> OrigenResult<T>
596    where
597        F: FnMut(Option<&PathBuf>) -> OrigenResult<T>,
598    {
599        let dir = self.reference_dir();
600        if change_dir && dir.is_some() {
601            let dir = dir.unwrap();
602            Ok(with_dir(&dir, || Ok(f(Some(&dir))?))?)
603        } else {
604            f(dir.as_ref())
605        }
606    }

609pub fn search_for_from_pwd(paths: Vec<&str>, searching_for_app: bool) -> (bool, PathBuf) {
610    let base = env::current_dir();
611    let base = match base {
612        Ok(p) => p,
613        Err(_e) => {
614            return (false, PathBuf::new());
615        }
616    };
617    search_for(paths, searching_for_app, &base)
618}

620pub fn search_for(paths: Vec<&str>, searching_for_app: bool, base: &Path) -> (bool, PathBuf) {
621    if searching_for_app {
622        log_trace!("Searching for app");
623    }
624    let mut aborted = false;
625    let mut base = base.to_path_buf();
626
627    while !paths
628        .iter()
629        .fold(base.clone(), |acc, p| acc.join(p))
630        .is_file()
631        && !aborted
632    {
633        if !base.pop() {
634            aborted = true;
635        }
636    }
637
638    if aborted {
639        if searching_for_app {
640            log_debug!("No app found");
641        }
642        (false, PathBuf::new())
643    } else {
644        if searching_for_app {
645            log_debug!("App found at '{}'", base.display());
646        }
647        (true, base)
648    }
649}

651pub fn get_home_dir() -> PathBuf {
652    if cfg!(windows) {
653        PathBuf::from(env::var("USERPROFILE").expect("Please set environment variable USERPROFILE to point to your home directory, then try again"))
654    } else {
655        PathBuf::from(env::var("HOME").expect(
656            "Please set environment variable HOME to point to your home directory, then try again",
657        ))
658    }

661// Convert any paths with / to \ on Windows
662fn clean_path(path: &Path) -> PathBuf {
663    let clean_path;
664    if cfg!(target_os = "windows") {
665        if let Some(p) = path.to_str() {
666            let win_path = PathBuf::from_slash(p);
667            clean_path = win_path;
668        } else {
669            clean_path = path.to_path_buf();
670        }
671    } else {
672        clean_path = path.to_path_buf();
673    }
674    clean_path
675}
