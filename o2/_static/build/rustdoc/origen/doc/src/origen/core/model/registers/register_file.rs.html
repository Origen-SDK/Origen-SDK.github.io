7#[derive(Debug)]
8/// Represents a groups of registers within an address block. RegisterFiles can also contain
9/// other RegisterFiles.
10pub struct RegisterFile {
11    pub id: usize,
12    pub address_block_id: usize,
13    /// Optional, if this register file is a child of another register file then its parent ID
14    /// will be recorded here
15    pub register_file_id: Option<usize>,
16    pub name: String,
17    pub description: String,
18    // TODO: What is this?!
19    /// The dimension of the register, defaults to 1.
20    pub dim: u32,
21    /// The address offset from the containing address block or register file,
22    /// expressed in address_unit_bits from the parent memory map.
23    pub offset: u128,
24    /// The number of addressable units in the register file.
25    pub range: u64,
26    pub registers: IndexMap<String, usize>,
27    pub register_files: IndexMap<String, usize>,
28}
29
30impl Default for RegisterFile {
31    fn default() -> RegisterFile {
32        RegisterFile {
33            id: 0,
34            address_block_id: 0,
35            register_file_id: None,
36            name: "default".to_string(),
37            description: "".to_string(),
38            dim: 1,
39            offset: 0,
40            range: 0,
41            registers: IndexMap::new(),
42            register_files: IndexMap::new(),
43        }
44    }
45}
46
47impl RegisterFile {
48    /// Get the ID from the given register name
49    pub fn get_register_id(&self, name: &str) -> Result<usize> {
50        match self.registers.get(name) {
51            Some(x) => Ok(*x),
52            None => {
53                bail!(
54                    "The register file '{}' does not have a register named '{}'",
55                    self.name,
56                    name
57                )
58            }
59        }
60    }
61
62    /// Returns an immutable reference to the address block object that owns the register file.
63    /// Note that this may or may not be the immediate parent of the register file depending on
64    /// whether it is instantiated within another register file or not.
65    pub fn address_block<'a>(&self, dut: &'a MutexGuard<Dut>) -> Result<&'a AddressBlock> {
66        dut.get_address_block(self.address_block_id)
67    }
68
69    /// Returns an immutable reference to the register file object that owns the register file.
70    /// If it returns None it means that the register file is instantiated directly within an
71    /// address block.
72    pub fn register_file<'a>(&self, dut: &'a MutexGuard<Dut>) -> Option<Result<&'a RegisterFile>> {
73        match self.register_file_id {
74            Some(x) => Some(dut.get_register_file(x)),
75            None => None,
76        }
77    }
78
79    /// Returns the address_unit_bits size that the register file's offset is defined in.
80    pub fn address_unit_bits(&self, dut: &MutexGuard<Dut>) -> Result<u32> {
81        match self.register_file(dut) {
82            Some(x) => Ok(x?.address_unit_bits(dut)?),
83            None => Ok(self.address_block(dut)?.address_unit_bits(dut)?),
84        }
85    }
86
87    /// Returns the fully-resolved address taking into account all base addresses defined by the parent hierachy.
88    /// The returned address is with an address_unit_bits size of 1.
89    pub fn bit_address(&self, dut: &MutexGuard<Dut>) -> Result<u128> {
90        let base = match self.register_file(dut) {
91            Some(x) => x?.bit_address(dut)?,
92            None => self.address_block(dut)?.bit_address(dut)?,
93        };
94        Ok(base + (self.offset * self.address_unit_bits(dut)? as u128))
95    }
96
97    /// Returns a path to this register file like "dut.my_block.my_map.my_address_block.my_reg_file", but the map and address block portions
98    /// will be inhibited when they are 'default'. This is to keep map and address block concerns out of the view of users who
99    /// don't use them and simply define regs at the top-level of the block.
100    pub fn friendly_path(&self, dut: &MutexGuard<Dut>) -> Result<String> {
101        let path = match self.register_file(dut) {
102            Some(x) => x?.friendly_path(dut)?,
103            None => self.address_block(dut)?.friendly_path(dut)?,
104        };
105        if self.name == "default" {
106            Ok(path)
107        } else {
108            Ok(format!("{}.{}", path, self.name))
109        }
110    }
111}
