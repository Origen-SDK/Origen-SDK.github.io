31impl VectorBased for SMT8 {
32    fn comment_str(&self) -> &str {
33        "#"
34    }
35
36    fn file_ext(&self) -> &str {
37        "avc"
38    }
39
40    fn print_vector(
41        &self,
42        renderer: &mut Renderer,
43        repeat: u32,
44        _compressable: bool,
45    ) -> Option<Result<String>> {
46        Some(Ok(format!(
47            "R{} {} {} # <EoL Comment>;",
48            repeat,
49            {
50                match renderer.timeset_name() {
51                    Ok(s) => s,
52                    Err(e) => return Some(Err(e)),
53                }
54            },
55            // The pin states should have been previously updated from the PinAction node, or just has default values
56            {
57                match renderer.render_states() {
58                    Ok(s) => s,
59                    Err(e) => return Some(Err(e)),
60                }
61            }
62        )))
63    }
64
65    fn print_pinlist(&self, renderer: &mut Renderer) -> Option<Result<String>> {
66        let dut = DUT.lock().unwrap();
67        let pins = renderer.states(&dut).names().join(" ");
68        Some(Ok(format!("FORMAT {};", pins)))
69    }
70
71    fn print_pattern_end(&self, _renderer: &mut Renderer) -> Option<Result<String>> {
72        Some(Ok("SQPG STOP;".to_string()))
73    }
74
75    fn override_node(
76        &self,
77        renderer: &mut Renderer,
78        node: &Node<PAT>,
79    ) -> Option<Result<Return<PAT>>> {
80        match &node.attrs {
81            PAT::Capture(capture, _metadata) => {
82                if let Ok(ids) = capture.enabled_capture_pins() {
83                    for pin in ids.iter() {
84                        if let Some(_) = capture.symbol.as_ref() {
85                            renderer
86                                .capturing
87                                .insert(Some(*pin), capture.symbol.clone());
88                        } else {
89                            renderer.capturing.insert(
90                                Some(*pin),
91                                Some(crate::standards::actions::CAPTURE.to_string()),
92                            );
93                        }
94                    }
95                    Some(Ok(Return::Unmodified))
96                } else {
97                    None
98                }
99            }
100            _ => None,
101        }
102    }
103}
