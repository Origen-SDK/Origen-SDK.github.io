7pub fn define(py: Python, m: &PyModule) -> PyResult<()> {
8    let subm = PyModule::new(py, "mailer")?;
9    subm.add_wrapped(wrap_pyfunction!(boot_mailer))?;
10    subm.add_wrapped(wrap_pyfunction!(boot_maillists))?;
11    m.add_submodule(subm)?;
12    Ok(())
13}

15#[pyfunction]
16pub(crate) fn boot_mailer(py: Python) -> PyResult<Option<PyObject>> {
17    init_mailer(py).or_else(|e| {
18        log_error!("Unable to initialize mailer:");
19        log_error!("{}", e.to_string());
20        Ok(None)
21    })
22}

24pub(crate) fn init_mailer(py: Python) -> PyResult<Option<PyObject>> {
25    if let Some(mc) = &origen::ORIGEN_CONFIG.mailer {
26        log_trace!("Booting Mailer from Origen config...");
27        let mailer_obj = new_py_obj(
28            py,
29            pytype_from_str(
30                py,
31                mc.class.as_ref().map_or(OM_MAILER_CLASS_QP, |c| c.as_str()),
32            )?,
33            Some(Mailer::toml_config_into_args(py, mc)?),
34            None,
35        )?;
36        log_trace!("... Done!");
37        Ok(Some(mailer_obj.to_object(py)))
38    } else {
39        log_trace!("No mailer configuration found!");
40        Ok(None)
41    }
42}

44#[pyfunction]
45pub(crate) fn boot_maillists(py: Python) -> PyResult<Option<PyObject>> {
46    match init_maillists(py) {
47        Ok(mls_obj) => Ok(Some(mls_obj)),
48        Err(e) => {
49            log_error!("Unable to initialize maillists:");
50            log_error!("{}", e.to_string());
51            Ok(None)
52        }
53    }
54}

56pub(crate) fn init_maillists(py: Python) -> PyResult<PyObject> {
57    let mut default_dirs: Vec<String> = vec![];
58    let mut mls_config: MaillistsTOMLConfig;
59
60    // Check for maillists in the install directory
61    if let Some(path) = &origen::STATUS.cli_location() {
62        default_dirs.push(path.parent().unwrap().display().to_string());
63    }
64
65    if let Some(app) = &origen::STATUS.app {
66        let mut d = app.root.clone();
67        d.push("config");
68        if d.exists() {
69            default_dirs.push(d.display().to_string());
70            d.push("maillists");
71            if d.exists() {
72                default_dirs.push(d.display().to_string());
73            }
74        }
75    }
76
77    if let Some(config) = &origen::ORIGEN_CONFIG.maillists {
78        log_trace!("Booting Maillists from Origen config...");
79        mls_config = config.clone();
80    } else {
81        log_trace!("Booting Maillists from Origen defaults...");
82        mls_config = MaillistsTOMLConfig::default();
83    }
84    if let Some(dirs) = mls_config.directories {
85        default_dirs.extend(dirs);
86    }
87    mls_config.directories = Some(default_dirs);
88
89    let args_kwargs = Maillists::toml_config_into_args(py, "origen", Some(true), &mls_config)?;
90
91    let mls_obj = new_py_obj(
92        py,
93        pytype_from_str(
94            py,
95            mls_config
96                .class
97                .as_ref()
98                .map_or(OM_MAILLISTS_CLASS_QP, |c| c.as_str()),
99        )?,
100        Some(args_kwargs.0),
101        Some(args_kwargs.1),
102    )?;
103    log_trace!("... Done!");
104    Ok(mls_obj.to_object(py))
105}
