12pub fn define(py: Python, m: &PyModule) -> PyResult<()> {
13    let subm = PyModule::new(py, "unit_testers")?;
14    subm.add_class::<RunResult>()?;
15    subm.add_wrapped(wrap_pyfunction!(app_unit_tester))?;
16    m.add_submodule(subm)?;
17    Ok(())
18}

21#[pyclass(subclass)]
22pub struct RunResult {
23    // Origen Run Result
24    pub orr: Option<Outcome>,
25}
26
27#[pymethods]
28impl RunResult {
29    #[classmethod]
30    fn __init__(
31        _cls: &PyType,
32        instance: &PyAny,
33        passed: bool,
34        output: Option<String>,
35    ) -> PyResult<()> {
36        let mut i = instance.extract::<PyRefMut<Self>>()?;
37        i.orr = Some({
38            let mut o = Outcome::new_pass_or_fail(passed);
39            if let Some(out) = output {
40                o.set_msg(out);
41            }
42            o.subtype = Some(OutcomeSubtypes::UnitTestStatus);
43            o
44        });
45        Ok(())
46    }
47
48    #[new]
49    fn new() -> Self {
50        Self { orr: None }
51    }
52
53    #[getter]
54    fn passed(&self) -> PyResult<bool> {
55        Ok(self.get_orr()?.passed())
56    }
57
58    #[getter]
59    fn failed(&self) -> PyResult<bool> {
60        Ok(!self.passed()?)
61    }
62}
63
64impl RunResult {
65    fn get_orr(&self) -> PyResult<&Outcome> {
66        match self.orr.as_ref() {
67            Some(r) => Ok(r),
68            None => runtime_error!("UnitTest Result has not been fully initialized yet!"),
69        }
70    }
71}

73// Returns a new pytest driver with the parameters from the app config
74pub fn new_pytest_driver(py_config: &PyDict) -> PyResult<PyObject> {
75    Python::with_gil(|py| {
76        let locals = PyDict::new(py);
77        locals.set_item("py_config", py_config)?;
78        locals.set_item(
79            "origen_pytester",
80            py.import("origen.utility.unit_testers.pytest")?
81                .to_object(py),
82        )?;
83        let pytester = py.eval("origen_pytester.PyTest(py_config)", Some(locals), None)?;
84        Ok(pytester.to_object(py))
85    })
86}

88/// Creates a unit test driver from the application's ``config.toml``
89#[pyfunction]
90fn app_unit_tester() -> PyResult<Option<PyObject>> {
91    // Raise an error if we aren't in an application instance
92    let app;
93    match &STATUS.app {
94        Some(a) => app = a,
95        None => {
96            return runtime_error!(
97                "Cannot retrieve the application's unit test config: no application found!"
98            )
99        }
100    }
101
102    let config = app.config();
103    Python::with_gil(|py| {
104        if let Some(ut_config) = &config.unit_tester {
105            if let Some(c) = ut_config.get("system") {
106                // Get the module and try to import it
107                let split = c.rsplitn(1, ".");
108                if split.count() == 2 {
109                    // Have a class (hopefully) of the form 'a.b.Class'
110                    // let py_config = hashmap_to_pydict(py, ut_config)?;
111                    // return runtime_error!("custom unit tester not supported yet!");
112                    todo!();
113                } else {
114                    // fall back to some enumerated systems
115                    if &c.to_lowercase() == "pytest" {
116                        let py_config = map_to_pydict(py, &mut ut_config.iter())?;
117                        Ok(Some(new_pytest_driver(py_config.as_ref(py))?))
118                    } else if &c.to_lowercase() == "none" {
119                        Ok(None)
120                    } else {
121                        return runtime_error!(format!("Unrecognized unit tester system '{}'", c));
122                    }
123                }
124            } else {
125                // Invalid config
126                return runtime_error!("Could not discern unit tester from app config");
127            }
128        } else {
129            let temp = HashMap::<String, String>::new();
130            let py_config = map_to_pydict(py, &mut temp.iter())?;
131            Ok(Some(new_pytest_driver(py_config.as_ref(py))?))
132        }
133    })
134}
