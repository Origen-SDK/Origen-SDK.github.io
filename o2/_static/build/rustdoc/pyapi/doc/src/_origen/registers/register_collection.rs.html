5/// Implements the user API to work with a collection of registers. The collection could be associated
6/// with another container object (an address block or register file), or this could be its own collection
7/// of otherwise un-related registers.
8#[pyclass]
9#[derive(Debug, Clone)]
10pub struct RegisterCollection {
11    /// The ID of the address block which contains these registers. It is optional so that
12    /// an empty RegisterCollection can be created, or an ad-hoc collection registers
13    pub address_block_id: Option<usize>,
14    /// The ID of the register file which contains these registers. It is optional as registers
15    /// can be instantiated in an address block directly and are not necessarily within a
16    /// register file
17    pub register_file_id: Option<usize>,
18    /// The IDs of the contained registers. If not present then the IDs will be derived from either
19    /// the associated register file or address block. If both are defined then the register file IDs
20    /// will be used.
21    pub ids: Option<Vec<usize>>,
22    /// Iterator index
23    pub i: usize,
24}
25
26impl RegisterCollection {
27    pub fn new() -> RegisterCollection {
28        RegisterCollection {
29            address_block_id: None,
30            register_file_id: None,
31            ids: None,
32            i: 0,
33        }
34    }
35}

44/// User API methods, available to both Rust and Python
45#[pymethods]
46impl RegisterCollection {
47    fn len(&self) -> PyResult<usize> {
48        if let Some(x) = &self.ids {
49            return Ok(x.len());
50        }
51        let dut = origen::dut();
52        if let Some(x) = self.register_file_id {
53            return Ok(dut.get_register_file(x)?.registers.len());
54        }
55        if let Some(x) = self.address_block_id {
56            return Ok(dut.get_address_block(x)?.registers.len());
57        }
58        Ok(0)
59    }
60
61    fn keys(&self) -> PyResult<Vec<String>> {
62        let dut = origen::dut();
63        if let Some(ids) = &self.ids {
64            let mut keys: Vec<String> = Vec::new();
65            for id in ids {
66                keys.push(dut.get_register(*id)?.name.clone());
67            }
68            return Ok(keys);
69        }
70        if let Some(x) = self.register_file_id {
71            let rf = dut.get_register_file(x)?;
72            let keys: Vec<String> = rf.registers.keys().map(|x| x.clone()).collect();
73            return Ok(keys);
74        }
75        if let Some(x) = self.address_block_id {
76            let ab = dut.get_address_block(x)?;
77            let keys: Vec<String> = ab.registers.keys().map(|x| x.clone()).collect();
78            return Ok(keys);
79        }
80        Ok(Vec::new())
81    }
82
83    fn values(&self) -> PyResult<Vec<BitCollection>> {
84        let dut = origen::dut();
85        if let Some(ids) = &self.ids {
86            let values: Vec<BitCollection> = ids
87                .iter()
88                .map(|x| BitCollection::from_reg_id(*x, &dut))
89                .collect();
90            return Ok(values);
91        }
92        if let Some(x) = self.register_file_id {
93            let rf = dut.get_register_file(x)?;
94            let values: Vec<BitCollection> = rf
95                .registers
96                .values()
97                .map(|x| BitCollection::from_reg_id(*x, &dut))
98                .collect();
99            return Ok(values);
100        }
101        if let Some(x) = self.address_block_id {
102            let ab = dut.get_address_block(x)?;
103            let values: Vec<BitCollection> = ab
104                .registers
105                .values()
106                .map(|x| BitCollection::from_reg_id(*x, &dut))
107                .collect();
108            return Ok(values);
109        }
110        Ok(Vec::new())
111    }
112
113    fn items(&self) -> PyResult<Vec<(String, BitCollection)>> {
114        let dut = origen::dut();
115        if let Some(ids) = &self.ids {
116            let mut items: Vec<(String, BitCollection)> = Vec::new();
117            for id in ids {
118                items.push((
119                    dut.get_register(*id)?.name.clone(),
120                    BitCollection::from_reg_id(*id, &dut),
121                ));
122            }
123            return Ok(items);
124        }
125        if let Some(x) = self.register_file_id {
126            let rf = dut.get_register_file(x)?;
127            let items: Vec<(String, BitCollection)> = rf
128                .registers
129                .iter()
130                .map(|(k, v)| (k.to_string(), BitCollection::from_reg_id(*v, &dut)))
131                .collect();
132            return Ok(items);
133        }
134        if let Some(x) = self.address_block_id {
135            let ab = dut.get_address_block(x)?;
136            let items: Vec<(String, BitCollection)> = ab
137                .registers
138                .iter()
139                .map(|(k, v)| (k.to_string(), BitCollection::from_reg_id(*v, &dut)))
140                .collect();
141            return Ok(items);
142        }
143        Ok(Vec::new())
144    }
145
146    fn __repr__(&self) -> PyResult<String> {
147        Ok(format!("<RegisterCollection: {:?}>", self.keys().unwrap()))
148    }
149
150    fn __contains__(&self, name: &str) -> PyResult<bool> {
151        match self.__getitem__(name) {
152            Ok(_) => Ok(true),
153            Err(_) => Ok(false),
154        }
155    }
156
157    fn __getitem__(&self, name: &str) -> PyResult<BitCollection> {
158        let dut = origen::dut();
159        if let Some(ids) = &self.ids {
160            for id in ids {
161                let reg = dut.get_register(*id)?;
162                if reg.name == name {
163                    return Ok(BitCollection::from_reg_id(*id, &dut));
164                }
165            }
166            return Err(PyKeyError::new_err(format!(
167                "The register collection does not contain a register named '{}'",
168                name
169            )));
170        }
171        if let Some(x) = self.register_file_id {
172            let rf = dut.get_register_file(x)?;
173            let id = rf.get_register_id(name)?;
174            return Ok(BitCollection::from_reg_id(id, &dut));
175        }
176        if let Some(x) = self.address_block_id {
177            let ab = dut.get_address_block(x)?;
178            let id = ab.get_register_id(name)?;
179            return Ok(BitCollection::from_reg_id(id, &dut));
180        }
181        Err(PyKeyError::new_err(format!(
182            "Register collection does not contain a register named '{}'",
183            name
184        )))
185    }
186
187    fn __len__(&self) -> PyResult<usize> {
188        Ok(self.len()?)
189    }
