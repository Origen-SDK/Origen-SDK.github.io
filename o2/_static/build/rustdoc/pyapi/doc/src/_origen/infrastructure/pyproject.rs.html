1use origen::core::status::DependencySrc;
2use origen::STATUS;
3use pyo3::prelude::*;
4use pyo3::types::PyDict;
5
6pub fn populate_status(py: Python, status: &PyDict) -> PyResult<()> {
7    if let Some(d) = STATUS.dependency_src().as_ref() {
8        if let Some(pyproject) = d.src_file() {
9            status.set_item("pyproject", pyapi_metal::pypath!(py, pyproject.display()))?;
10        } else {
11            status.set_item("pyproject", py.None())?;
12        }
13        status.set_item("invocation", PyProjectSrc::from(d).to_py(py)?)?;
14    } else {
15        status.set_item("pyproject", py.None())?;
16        status.set_item("invocation", py.None())?;
17    }
18    Ok(())
19}
20
21pub fn define(py: Python, m: &PyModule) -> PyResult<()> {
22    let subm = PyModule::new(py, "pyproject")?;
23    subm.add_class::<PyProjectSrc>()?;
24    m.add_submodule(subm)?;
25    Ok(())
26}

28#[pyclass]
29pub enum PyProjectSrc {

32    UserGlobal,
33    Global,
34    NoneFound,

36
37impl PyProjectSrc {
38    pub fn to_py(self, py: Python) -> PyResult<Py<Self>> {
39        Py::new(py, self)
40    }
41}
42
43impl From<&DependencySrc> for PyProjectSrc {
44    fn from(src: &DependencySrc) -> Self {
45        match src {
46            DependencySrc::App(_) => Self::App,
47            DependencySrc::Workspace(_) => Self::Workspace,
48            DependencySrc::UserGlobal(_) => Self::UserGlobal,
49            DependencySrc::Global(_) => Self::Global,
50            DependencySrc::NoneFound => Self::NoneFound,
51        }
52    }
