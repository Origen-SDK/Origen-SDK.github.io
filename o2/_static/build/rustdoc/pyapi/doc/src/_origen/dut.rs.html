17    let subm = PyModule::new(py, "dut")?;

26/// The PyDUT object, through which DUT-related interactions between the Python frontend and the Rust backend take place.
27#[pyclass]
28#[derive(Debug)]
29pub struct PyDUT {
30    metadata: Vec<PyObject>,
31}
32
33#[pymethods]
34impl PyDUT {
35    #[new]
36    /// Instantiating a new instance of PyDUT means re-loading the target
37    fn new(name: &str) -> PyResult<Self> {
38        origen::dut().change(name)?;
39        origen::services().change();
40        Ok(PyDUT { metadata: vec![] })
41    }
42
43    /// Creates a new model at the given path
44    #[pyo3(signature=(parent_id, name, offset=None))]
45    fn create_model(
46        &self,
47        parent_id: Option<usize>,
48        name: &str,
49        offset: Option<u128>,
50    ) -> PyResult<usize> {
51        Ok(origen::dut().create_model(parent_id, name, offset)?)
52    }
53
54    fn model_console_display(&self, model_id: usize) -> PyResult<String> {
55        let dut = origen::dut();
56        let model = dut.get_model(model_id)?;
57        Ok(model.console_display(&dut)?)
58    }
59
60    /// push_metadata(self, item)
61    /// Pushes metadata object onto the current DUT
62    pub fn push_metadata(&mut self, py: Python, item: &PyAny) -> usize {
63        self.metadata.push(item.to_object(py));
64        self.metadata.len() - 1
65    }
66
67    pub fn override_metadata_at(&mut self, py: Python, idx: usize, item: &PyAny) -> PyResult<()> {
68        if self.metadata.len() > idx {
69            self.metadata[idx] = item.to_object(py);
70            Ok(())
71        } else {
72            Err(PyErr::from(Error::new(&format!(
73                "Overriding metadata at {} exceeds the size of the current metadata vector!",
74                idx
75            ))))
76        }
77    }
78
79    pub fn get_metadata(&self, idx: usize) -> PyResult<&PyObject> {
80        Ok(&self.metadata[idx])
81    }
82
83    pub fn model(&self, id: usize) -> PyResult<ModelProxy> {
84        Ok(ModelProxy::new(id))
85    }
86
87    pub fn empty_regs(&self) -> PyResult<RegisterCollection> {
88        Ok(RegisterCollection::new())
89    }
90}

92impl PyDUT {
93    pub fn ensure_pins(model_path: &str) -> PyResult<()> {
94        Python::with_gil(|py| {
95            let locals = PyDict::new(py);
96            locals.set_item("origen", py.import("origen")?.to_object(py))?;
97            py.eval(&format!("origen.{}.pins", model_path), Some(locals), None)?;
98            Ok(())
99        })
100    }
