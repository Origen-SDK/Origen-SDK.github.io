5pub fn add_core_subc_helps(
6    helps: &mut CmdHelps,
7    base_name: &str,
8    cmd: &str,
9    cmd_help: &str,
10    extendable: bool,
11    subcmds: &[SubCmd],
12) {
13    let n = format!("{}.{}", base_name, cmd);
14    let h = helps.add_core_cmd(&n).set_help_msg(cmd_help);
15    if !extendable {
16        h.set_as_not_extendable();
17    }
18    for subc in subcmds {
19        add_core_subc_helps(
20            helps,
21            &n,
22            subc.name,
23            subc.help,
24            subc.extendable,
25            subc.subcmds,
26        );
27    }
28}

30pub fn add_core_subcs<'a>(
31    helps: &'a CmdHelps,
32    exts: Option<&'a Extensions>,
33    cmd: Command,
34    base: Vec<&str>,
35    subcmd: &SubCmd,
36) -> Result<Command> {
37    let mut n = base.clone();
38    n.push(subcmd.name);
39    let mut subc = helps.core_subc(&n);
40    for s in subcmd.subcmds {
41        subc = add_core_subcs(helps, exts, subc, n.clone(), s)?;
42    }
43    if let Some(setup) = subcmd.proc {
44        subc = setup(subc);
45    }
46    if subcmd.include_app_opts {
47        subc = super::add_all_app_opts(subc);
48    }
49    // add exts
50    if let Some(e) = exts {
51        subc = e.apply_to_core_cmd(&n.join("."), subc);
52    }
53    Ok(cmd.subcommand(subc))
54}
