1//! Helpers for the raw tester expressions used by IfExpr and UnlessExpr flow conditions
2
3/// Returns the flow variables read by the given tester expression.
4///
5/// Literals (numbers, strings, true/false), the operators and/or/not, function names and
6/// the parts of dotted references (e.g. a sub-flow's variable) are not flow variables of
7/// the current flow.
8pub fn expression_variables(expression: &str) -> Vec<String> {
9    let chars: Vec<char> = expression.chars().collect();
10    let mut vars: Vec<String> = vec![];
11    let mut i = 0;
12    while i < chars.len() {
13        let c = chars[i];
14        if c == '"' || c == '\'' {
15            i += 1;
16            while i < chars.len() && chars[i] != c {
17                if chars[i] == '\\' {
18                    i += 1;
19                }
20                i += 1;
21            }
22            i += 1;
23        } else if c.is_ascii_digit() {
24            while i < chars.len()
25                && (chars[i].is_ascii_alphanumeric() || chars[i] == '_' || chars[i] == '.')
26            {
27                i += 1;
28            }
29        } else if c.is_ascii_alphabetic() || c == '_' {
30            let start = i;
31            while i < chars.len() && (chars[i].is_ascii_alphanumeric() || chars[i] == '_') {
32                i += 1;
33            }
34            let name: String = chars[start..i].iter().collect();
35            let prev = chars[..start].iter().rev().find(|c| !c.is_whitespace());
36            let next = chars[i..].iter().find(|c| !c.is_whitespace());
37            let is_literal = name == "true" || name == "false";
38            let is_operator = name == "and" || name == "or" || name == "not";
39            let is_dotted = prev == Some(&'.') || next == Some(&'.');
40            let is_function = next == Some(&'(');
41            if !is_literal && !is_operator && !is_dotted && !is_function && !vars.contains(&name) {
42                vars.push(name);
43            }
44        } else {
45            i += 1;
46        }
47    }
48    vars
49}

51#[cfg(test)]
52mod tests {
53    use super::expression_variables;
54
55    #[test]
56    fn expression_variables_excludes_literals_functions_and_references() {
57        assert_eq!(expression_variables("mode == 6"), vec!["mode"]);
58        assert_eq!(
59            expression_variables("captureMode == 6 && (Other_var != 1.5e3)"),
60            vec!["captureMode", "Other_var"]
61        );
62        assert_eq!(
63            expression_variables("abs(delta) > 2 && sub.result == true && name == \"a b\""),
64            vec!["delta", "name"]
65        );
66        assert_eq!(expression_variables("count + count"), vec!["count"]);
67    }
68
69    #[test]
70    fn expression_variables_excludes_operators() {
71        assert_eq!(
72            expression_variables("A == 1 or not (B and C)"),
73            vec!["A", "B", "C"]
74        );
75    }
