1//! The processor API is intentionally placed in a separate modele from the AST/Node
2//! to ensure that processor implementations use the Node API rather than coupling to its
3//! internals (i.e. children vector) which could be subject to change.
4
5//use crate::generator::ast::*;
6use crate::ast::node::Node;
7pub use crate::Result;
8
9/// All procesor handler methods should return this
10pub enum Return<T> {
11    /// Deletes the node from the output AST.
12    None,
13    /// Clones the node (and all of its children) into the output AST. Note that
14    /// the child nodes are not processed in this case (though they will appear in
15    /// the output unmodified).
16    Unmodified,
17    /// Clones the node but replaces it's current children with their
18    /// processed counterparts in the output AST.
19    ProcessChildren,
20    /// Replace the node in the output AST with the given node.
21    Replace(Node<T>),
22    /// Removes the node and leaves its children in its place.
23    Unwrap,
24    /// Replace the node in the output AST with the given nodes, the vector wrapper
25    /// will be removed and the nodes will be placed inline with where the current
26    /// node is/was.
27    Inline(Vec<Node<T>>),
28    /// Same as Inline, but accepts a vector of boxed nodes
29    InlineBoxed(Vec<Box<Node<T>>>),
30    /// A combination of Unwrap and ProcessChildren, which will unwrap the current node
31    /// but leave processed children in its place
32    UnwrapWithProcessedChildren,
33    /// A combination of Inline and ProcessChildren which will add the given nodes
34    /// then proceed to process the original node's children
35    InlineWithProcessedChildren(Vec<Node<T>>),
36    /// Keep the node in the new AST but replace its children with the given nodes
37    ReplaceChildren(Vec<Node<T>>),
38}
39
40pub trait Processor<T> {
41    fn on_node(&mut self, _node: &Node<T>) -> Result<Return<T>> {
42        Ok(Return::ProcessChildren)
43    }
44
45    /// This will be called at the end of processing every node which has children.
46    /// The node which is about to be closed is provided in the arguments.
47    /// Note that you should probably never return a derivative of the given node
48    /// here, it should either be None or a new node(s)
49    fn on_end_of_block(&mut self, _node: &Node<T>) -> Result<Return<T>> {
50        Ok(Return::None)
51    }
52
53    fn on_processed_node(&mut self, _node: &Node<T>) -> Result<Return<T>> {
54        Ok(Return::Unmodified)
55    }
56}
