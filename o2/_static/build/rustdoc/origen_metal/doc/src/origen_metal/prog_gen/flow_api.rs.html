368/// Returns an error if the given name is not an identifier, or if the current position is not at
369/// flow level, e.g. it is within a condition or a loop
370fn ensure_flow_level_declaration(api: &str, name: &str) -> Result<()> {
371    if !is_identifier(&name) {
372        bail!(
373            "{}('{}'): the name of a flow variable must be an identifier \
374             (letters, digits and underscores, not starting with a digit)",
375            api,
376            name
377        );
378    }
379    FLOW.with_selected_flow(|flow| {
380        for attrs in flow.open_attrs() {
381            match attrs {
382                PGM::Flow(_) | PGM::SubFlow(_, _) => break,
383                PGM::Group(_, _, kind, _) if kind == &GroupType::Flow => break,
384                PGM::Condition(_)
385                | PGM::Loop(..)
386                | PGM::OnFailed(_)
387                | PGM::OnPassed(_)
388                | PGM::Else
389                | PGM::Whenever
390                | PGM::WheneverAny
391                | PGM::WheneverAll => bail!(
392                    "{}('{}') must be called at flow level, not inside a condition or loop",
393                    api,
394                    name
395                ),
396                _ => {}
397            }
398        }
399        Ok(())
400    })
401}

403/// Returns true if the given name can be used as a flow variable name: letters, digits and
404/// underscores, not starting with a digit
405pub fn is_identifier(name: &str) -> bool {
406    let mut chars = name.chars();
407    match chars.next() {
408        Some(c) if c.is_ascii_alphabetic() || c == '_' => {}
409        _ => return false,
410    }
411    chars.all(|c| c.is_ascii_alphanumeric() || c == '_')
412}

414/// Start a loop which repeats its contents `count` times, the returned reference should be
415/// retained and passed to end_block.
416/// The count is either fixed or the name of a flow variable which holds it when the loop starts.
417/// A fixed count must be at least 1, and so must a count held in a variable when the loop
418/// starts: like a hand-written V93K SMT8 `for` loop, the range is not guarded and a lower count
419/// counts down instead.
420/// The iteration number is available within the loop as the given variable, counting up from
421/// `start` through `count` values, or down to `start` when `descending` is true, e.g. 1 to 3,
422/// or 0 to 2 when `start` is 0, for a count of 3.
423pub fn start_loop(
424    count: Option<LoopCount>,
425    variable: Option<String>,
426    descending: bool,
427    start: i64,
428    meta: Option<Meta>,
429) -> Result<usize> {
430    if let Some(LoopCount::Fixed(0)) = &count {
431        bail!("loop count 0: a loop must repeat at least once");
432    }
433    if let Some(LoopCount::Variable(name)) = &count {
434        if !is_identifier(name) {
435            bail!(
436                "loop count '{}': a loop count must be a number or the name of a flow variable \
437                 (letters, digits and underscores, not starting with a digit)",
438                name
439            );
440        }
441    }
442    let n = node!(PGM::Loop, count, variable, descending, start; meta);
443    FLOW.push_and_open(n)
444}
