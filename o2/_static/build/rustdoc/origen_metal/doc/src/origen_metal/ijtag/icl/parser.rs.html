14pub(crate) fn parse_file_with_options(path: &Path, options: ParseOptions) -> Result<Node<ICL>> {
15    if !path.exists() {
16        return Err(Error::new(&format!(
17            "File does not exist: {}",
18            path.display()
19        )));
20    }
21
22    let expanded = source::from_file(path)?;
23    parse_expanded_with_options(
24        &expanded.text,
25        Some(&expanded.entry),
26        Some(&expanded.map),
27        options,
28    )
29    .map_err(|e| {
30        Error::new(&format!(
31            "Error parsing file {}:\n{}",
32            expanded.entry, e.msg
33        ))
34    })
35}

37pub(crate) fn parse_str_with_options(
38    icl: &str,
39    source_file: Option<&str>,
40    options: ParseOptions,
41) -> Result<Node<ICL>> {
42    let expanded = source::from_str(icl, source_file)?;
43    parse_expanded_with_options(&expanded.text, source_file, Some(&expanded.map), options)

46fn parse_expanded_with_options(
47    icl: &str,
48    source_file: Option<&str>,
49    source_map: Option<&source::SourceMap>,
50    options: ParseOptions,
51) -> Result<Node<ICL>> {
52    match ICLParser::parse(Rule::icl_source, icl) {
53        Err(error) => {
54            let offset = match error.location {
55                pest::error::InputLocation::Pos(offset) => offset,
56                pest::error::InputLocation::Span((offset, _)) => offset,
57            };
58            let location = source_map
59                .map(|map| map.location(offset as u32, icl))
60                .unwrap_or_else(|| source_file.unwrap_or("<string>").to_string());
61            Err(Error::new(&format!("{location}:\n{error}")))
62        }
63        Ok(mut parsed) => to_ast(parsed.next().unwrap(), source_file, options),
64    }
65}

222fn leaf_or_transparent(pair: &Pair<'_, Rule>, options: ParseOptions) -> Action {
223    let text = pair.as_str();
224    match pair.as_rule() {
225        Rule::scalar_id => Action::Leaf(ICL::Identifier(text.to_string())),
226        Rule::parameter_ref => Action::Leaf(ICL::ParameterReference(text.to_string())),
227        Rule::string => Action::Leaf(ICL::StringLiteral(text.to_string())),
228        Rule::sized_bin_number
229        | Rule::sized_dec_number
230        | Rule::sized_hex_number
231        | Rule::unsized_bin_number
232        | Rule::unsized_dec_number
233        | Rule::unsized_hex_number
234        | Rule::pos_int => Action::Leaf(ICL::Number(text.to_string())),
235        Rule::time_unit => Action::Leaf(ICL::TimeUnit(text.to_string())),
236        Rule::iproc_argument => Action::Leaf(ICL::IProcArgument(text.to_string())),
237        Rule::generic_access_link_block => {
238            Action::Leaf(ICL::GenericAccessLinkBody(text.to_string()))
239        }
240        Rule::invert => Action::Leaf(ICL::Invert),
241        Rule::integer_add_op => {
242            if text == "+" {
243                Action::Leaf(ICL::Add)
244            } else {
245                Action::Leaf(ICL::Subtract)
246            }
247        }
248        Rule::integer_mul_op => match text {
249            "*" => Action::Leaf(ICL::Multiply),
250            "/" => Action::Leaf(ICL::Divide),
251            _ => Action::Leaf(ICL::Modulo),
252        },
253        Rule::logic_bool_op => {
254            if text == "&&" {
255                Action::Leaf(ICL::BooleanAnd)
256            } else {
257                Action::Leaf(ICL::BooleanOr)
258            }
259        }
260        Rule::logic_bitwise_op => match text {
261            "&" => Action::Leaf(ICL::BitwiseAnd),
262            "|" => Action::Leaf(ICL::BitwiseOr),
263            _ => Action::Leaf(ICL::BitwiseXor),
264        },
265        Rule::logic_equality_op => {
266            if text == "==" {
267                Action::Leaf(ICL::Equal)
268            } else {
269                Action::Leaf(ICL::NotEqual)
270            }
271        }
272        Rule::logic_unary_op => {
273            if text == "!" {
274                Action::Leaf(ICL::BooleanNot)
275            } else {
276                Action::Leaf(ICL::Invert)
277            }
278        }
279        Rule::logic_reduction_op => match text {
280            "&" => Action::Leaf(ICL::BitwiseAnd),
281            "|" => Action::Leaf(ICL::BitwiseOr),
282            _ => Action::Leaf(ICL::BitwiseXor),
283        },
284        Rule::COMMENT if options.preserve_comments => Action::Leaf(ICL::Comment(text.to_string())),
285        _ => Action::Transparent,
286    }
287}

289fn to_ast(
290    pair: Pair<'_, Rule>,
291    source_file: Option<&str>,
292    options: ParseOptions,
293) -> Result<Node<ICL>> {
294    debug_assert_eq!(pair.as_rule(), Rule::icl_source);
295    let mut ast = AST::new();
296    ast.push_and_open(node!(ICL::Root));
297    if let Some(file) = source_file {
298        ast.push(node!(ICL::SourceFile, file.to_string()));
299    }
300
301    let mut frames: Vec<(Pairs<'_, Rule>, Option<usize>)> = vec![(pair.into_inner(), None)];
302    while let Some((pairs, close_id)) = frames.last_mut() {
303        if let Some(next) = pairs.next() {
304            match action(&next, options) {
305                Action::Leaf(attrs) => ast.push(Node::new(attrs)),
306                Action::Open(attrs) => {
307                    let id = ast.push_and_open(Node::new(attrs));
308                    frames.push((next.into_inner(), Some(id)));
309                }
310                Action::Transparent => {
311                    let inner = next.into_inner();
312                    if inner.peek().is_some() {
313                        frames.push((inner, None));
314                    }
315                }
316                Action::Ignore => {}
317            }
318        } else {
319            let close_id = *close_id;
320            frames.pop();
321            if let Some(id) = close_id {
322                ast.close(id)?;
323            }
324        }
325    }
326
327    Ok(ast.unwrap())
328}

336    fn count(node: &Node<ICL>, predicate: &dyn Fn(&ICL) -> bool) -> usize {
337        usize::from(predicate(&node.attrs))
338            + node
339                .children
340                .iter()
341                .map(|child| count(child, predicate))
342                .sum::<usize>()
343    }
344
345    fn contains(node: &Node<ICL>, predicate: &dyn Fn(&ICL) -> bool) -> bool {
346        predicate(&node.attrs) || node.children.iter().any(|child| contains(child, predicate))
347    }
