7pub fn resolve_relative_paths_to_strings<D: Display>(
8    paths: &Vec<D>,
9    relative_to: &PathBuf,
10) -> Vec<String> {
11    let mut retn: Vec<String> = vec![];
12    for p in paths.iter() {
13        retn.push(to_abs_path(p, relative_to).display().to_string());
14    }
15    retn

18/// Given a relative path and a path its relative to, resolve the absolute path
19///
20/// In Windows, 'canonicalize' prepends "\\?\".
21/// However, upon some google searching, it apparently is
22/// dangerous to just remove as, in certain circumstances,
23/// it does resolve to a different pattern.
24/// But, other libraries, like glob, cannot handle it,
25/// even when the ? is escaped.
26/// So, instead just mashing the relative piece atop the
27/// relative_to directory and users of these paths
28/// can decide how to resolve.
29/// For example, glob has no problem correctly resolving
30/// this, without canonicalize
31///
32/// TL;DR - this needs some work in the future
33pub fn to_abs_path<D: Display>(path: D, relative_to: &PathBuf) -> PathBuf {
34    let pb = PathBuf::from(path.to_string());
35    if pb.is_relative() {
36        let mut resolved = PathBuf::from(relative_to);
37        resolved.push(pb);
38        resolved
39    } else {
40        pb
41    }
42}

44/// Render a potential template, or return the contents of the non-template file.
45pub fn preprocess_as_template(path: &PathBuf) -> Result<Option<String>> {
46    let mut entries = vec![];
47    for entry in glob::glob(&path.display().to_string())? {
48        match entry {
49            Ok(e) => entries.push(e),
50            Err(e) => bail!("Error reading entry: {}", e),
51        }
52    }
53    if entries.is_empty() {
54        Ok(None)
55    } else {
56        let name = super::resolve_os_str(path.file_name().unwrap_or(std::ffi::OsStr::new("")))?;
57        if entries.len() > 1 {
58            // Not supporting multiple paths or having some hierarchy
59            // Just report as an error
60            let mut entries_list = String::new();
61            for entry in &entries {
62                entries_list.push_str(&format!("  {:?}\n", entry));
63            }
64            bail!("Multiple matches found for '{}':\n{}", name, entries_list);
65        } else {
66            let file = entries.first().unwrap();
67            let content = fs::read_to_string(&file)?;
68
69            if file
70                .extension()
71                .map_or(false, |ext| ext == "j2" || ext == "jinja")
72            {
73                // Evaluate using minijinja
74                let mut env = minijinja::Environment::new();
75                let file_content = fs::read_to_string(&file)?;
76                env.add_template(&name, &file_content)?;
77                let tmpl = env.get_template(&name)?;
78                Ok(Some(tmpl.render("")?))
79            } else {
80                Ok(Some(content))
81            }
82        }
83    }
