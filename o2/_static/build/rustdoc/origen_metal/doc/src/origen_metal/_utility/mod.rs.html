10/// Checks the given values of a vector against an enumerated set of accepted values.
11/// Optionally, check for duplicate items as well.
12pub fn validate_input_list<'a, S, T, U>(
13    input: T,
14    allowed_values: Option<U>,
15    allow_duplicates: bool,
16    duplicate_error_customizer: Option<&dyn Fn(&S, usize, usize) -> String>,
17    value_error_customizer: Option<&dyn Fn(&Vec<&&S>) -> String>,
18) -> Result<()>
19where
20    S: 'a + std::cmp::Eq + std::hash::Hash + std::fmt::Display,
21    T: IntoIterator<Item = &'a S>,
22    U: IntoIterator<Item = &'a S>,
23{
24    let input_vector: Vec<&S> = input.into_iter().collect();
25    let mut input_hash: HashSet<&S> = HashSet::new();
26    for (idx, i) in input_vector.iter().enumerate() {
27        let added = input_hash.insert(i);
28        if !allow_duplicates & !added {
29            let f_idx = input_vector
30                .iter()
31                .position(|x| x == i)
32                .expect(&format!("Expected element {} in input", i.to_string()));
33            if let Some(f) = duplicate_error_customizer {
34                bail!(&f(i, f_idx, idx));
35            } else {
36                bail!(
37                    "Input contains duplicate value '{}' at index {}, which is not allowed in the this context (first occurrence at index {})",
38                    i.to_string(),
39                    idx,
40                    f_idx
41                );
42            }
43        }
44    }
45
46    if let Some(allowed) = allowed_values {
47        let hash_allowed = HashSet::from_iter(allowed.into_iter());
48        let diff: Vec<&&S> = input_hash.difference(&hash_allowed).into_iter().collect();
49        if diff.len() > 0 {
50            if let Some(f) = value_error_customizer {
51                bail!(&f(&diff));
52            } else {
53                bail!(
54                    "Input contains values which are not allowed in this context: {}",
55                    diff.iter()
56                        .map(|d| format!("'{}'", d))
57                        .collect::<Vec<String>>()
58                        .join(", ")
59                );
60            }
61        }
62    }
63    Ok(())
64}

66pub fn str_to_bool(s: &str) -> Result<bool> {
67    match s {
68        "true" | "True" => Ok(true),
69        "false" | "False" => Ok(false),
70        _ => bail!("Could not convert string {} to boolean value", s),
71    }
72}

74pub fn str_from_byte_array(bytes: &[u8]) -> Result<String> {
75    let mut s = String::new();
76    for b in bytes {
77        s.push_str(&format!("{:02x}", b));
78    }
79    Ok(s)
80}

82pub fn resolve_os_str(s: &OsStr) -> Result<String> {

86/// For convenience in converting to/from configs, allow bytes to be converted
87/// to a string representation of byte values (NOT characters themselves).
88/// This allows for invalid UTF-8 characters to still be encoded in a string.
89pub fn bytes_from_str_of_bytes(s: &str) -> Result<Vec<u8>> {
90    let mut bytes = vec![];
91    for chrs in s.chars().collect::<Vec<char>>().chunks(2) {
92        let s = format!("{}{}", chrs[0], chrs[1]);
93        bytes.push(
94            u8::from_str_radix(&s, 16)
95                .expect(&format!("Could not interpret byte {} as a hex value", s)),
96        );
97    }
98    Ok(bytes)
99}

101pub fn unsorted_dedup<T: Eq + std::hash::Hash + Clone>(v: &mut Vec<T>) {
102    let mut uniques = std::collections::HashSet::new();
103    v.retain(|e| uniques.insert(e.clone()));
