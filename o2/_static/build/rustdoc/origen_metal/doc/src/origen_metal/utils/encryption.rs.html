60pub fn encrypt_with(
61    data: &str,
62    key: GenericArray<u8, U32>,
63    nonce: GenericArray<u8, U12>,
64) -> Result<Vec<u8>> {
65    let cipher = Aes256Gcm::new(&key);
66    Ok(cipher.encrypt(&nonce, data.as_bytes())?)
67}

81pub fn decrypt_with(
82    data: &[u8],
83    key: GenericArray<u8, U32>,
84    nonce: GenericArray<u8, U12>,
85) -> Result<String> {
86    let cipher = Aes256Gcm::new(&key);
87    let plaintext = cipher.decrypt(&nonce, data)?;
88    Ok(String::from_utf8(plaintext)?)
89}

104    #[test]
105    fn test_str_byte_array_roundtrip() {
106        let s1 = "TEST";
107        let s2 = "another test";
108
109        // Get the actual byte values of the characters in the strings
110        let bytes1 = s1.as_bytes();
111        let bytes2 = s2.as_bytes();
112
113        // Convert the byte values to a string of 2-char hex values of the bytes
114        // e.g.: [1, 8, 16, 32] => "01081020"
115        let bs1 = str_from_byte_array(bytes1).unwrap();
116        let bs2 = str_from_byte_array(bytes2).unwrap();
117        assert_eq!(bs1, "54455354");
118        assert_eq!(bs2, "616e6f746865722074657374");
119
120        // Convert bytes string back to array of bytes
121        // e.g.: "01081020" => [1, 8, 16, 32]
122        let b1 = bytes_from_str_of_bytes(&bs1).unwrap();
123        let b2 = bytes_from_str_of_bytes(&bs2).unwrap();
124        assert_eq!(b1, [0x54, 0x45, 0x53, 0x54]);
125        assert_eq!(
126            b2,
127            [0x61, 0x6e, 0x6f, 0x74, 0x68, 0x65, 0x72, 0x20, 0x74, 0x65, 0x73, 0x74]
128        );
129
130        // Convert the byte values back to a str to check the roundtrip.
131        // Note: invalid utf-8 characters may arise when this is actually used for key generation,
132        // so this next step could fail.
133        let r1 = std::str::from_utf8(&b1).unwrap();
134        let r2 = std::str::from_utf8(&b2).unwrap();
135
136        assert_eq!(r1, s1);
137        assert_eq!(r2, s2);
138    }
