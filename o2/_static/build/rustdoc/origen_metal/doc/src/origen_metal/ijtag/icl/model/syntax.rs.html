315    pub fn load_or_elaborate(
316        &self,
317        source_path: &Path,
318        top: Option<&str>,
319        cache_dir: &Path,

323        if let Some(model) =

327        }

338    pub(super) fn parse_expanded(&self, expanded: ExpandedSource) -> Result<ParsedIcl> {
339        let source = expanded.text;
340        let source_file = Some(expanded.entry);
341        if source.len() > u32::MAX as usize {
342            return Err(Error::new(
343                "ICL sources larger than 4 GiB are not supported",
344            ));
345        }
346        let builder =
347            if self.threads > 1 && source.len() >= 1_000_000 && !contains_top_level_pdl(&source) {
348                build_parallel(&source, self.preserve_comments, self.threads)?
349            } else {
350                let pairs = ICLParser::parse(Rule::icl_source, &source).map_err(|error| {
351                    let offset = match error.location {
352                        pest::error::InputLocation::Pos(offset) => offset,
353                        pest::error::InputLocation::Span((offset, _)) => offset,
354                    };
355                    Error::new(&format!(
356                        "{}:\n{}",
357                        expanded.map.location(offset as u32, &source),
358                        error
359                    ))
360                })?;
361                build(&source, pairs, self.preserve_comments)?
362            };
363        Ok(ParsedIcl {
364            inner: Arc::new(ParsedStorage {
365                source,
366                source_file,
367                nodes: builder.nodes,
368                symbols: builder.symbols,
369                symbol_ids: builder.symbol_ids,
370                source_map: expanded.map,
371            }),
372        })
373    }

393struct Builder {
394    nodes: Vec<SyntaxNode>,
395    last_child: Vec<Option<SyntaxId>>,
396    symbols: Vec<String>,
397    symbol_ids: HashMap<String, SymbolId>,
398}
399
400impl Builder {
401    fn new(source_len: usize) -> Self {
402        Self {
403            nodes: vec![SyntaxNode {
404                kind: SyntaxKind::Root,
405                span: SourceSpan {
406                    start: 0,
407                    end: source_len as u32,
408                },
409                first_child: None,
410                next_sibling: None,
411            }],
412            last_child: vec![None],
413            symbols: Vec::new(),
414            symbol_ids: HashMap::new(),
415        }
416    }
417
418    fn intern(&mut self, value: &str) -> SymbolId {
419        if let Some(id) = self.symbol_ids.get(value) {
420            return *id;
421        }
422        let id = SymbolId(self.symbols.len() as u32);
423        let value = value.to_string();
424        self.symbols.push(value.clone());
425        self.symbol_ids.insert(value, id);
426        id
427    }
428
429    fn add(&mut self, parent: SyntaxId, kind: SyntaxKind, span: SourceSpan) -> SyntaxId {
430        let id = SyntaxId(self.nodes.len() as u32);
431        self.nodes.push(SyntaxNode {
432            kind,
433            span,
434            first_child: None,
435            next_sibling: None,
436        });
437        self.last_child.push(None);
438        if let Some(previous) = self.last_child[parent.as_usize()] {
439            self.nodes[previous.as_usize()].next_sibling = Some(id);
440        } else {
441            self.nodes[parent.as_usize()].first_child = Some(id);
442        }
443        self.last_child[parent.as_usize()] = Some(id);
444        id
445    }
446
447    fn merge(&mut self, other: Builder, offset: u32) {
448        let mut symbol_map = Vec::with_capacity(other.symbols.len());
449        for symbol in &other.symbols {
450            symbol_map.push(self.intern(symbol));
451        }
452        let base = self.nodes.len() as u32 - 1;
453        let remap_id = |id: SyntaxId| SyntaxId(base + id.0);
454        let remap_kind = |kind: SyntaxKind| match kind {
455            SyntaxKind::Identifier(symbol) => SyntaxKind::Identifier(symbol_map[symbol.as_usize()]),
456            SyntaxKind::ParameterReference(symbol) => {
457                SyntaxKind::ParameterReference(symbol_map[symbol.as_usize()])
458            }
459            other => other,
460        };
461
462        for node in other.nodes.iter().skip(1) {
463            self.nodes.push(SyntaxNode {
464                kind: remap_kind(node.kind),
465                span: SourceSpan {
466                    start: node.span.start + offset,
467                    end: node.span.end + offset,
468                },
469                first_child: node.first_child.map(remap_id),
470                next_sibling: node.next_sibling.map(remap_id),
471            });
472        }
473        for child in other.last_child.iter().skip(1) {
474            self.last_child.push(child.map(remap_id));
475        }
476
477        if let Some(local_first) = other.nodes[0].first_child {
478            let first = remap_id(local_first);
479            if let Some(previous) = self.last_child[0] {
480                self.nodes[previous.as_usize()].next_sibling = Some(first);
481            } else {
482                self.nodes[0].first_child = Some(first);
483            }
484            self.last_child[0] = other.last_child[0].map(remap_id);
485        }
486    }
487}

497fn build(source: &str, mut pairs: Pairs<'_, Rule>, preserve_comments: bool) -> Result<Builder> {
498    let root_pair = pairs.next().expect("successful ICL parse has a root pair");
499    let mut builder = Builder::new(source.len());
500    let mut frames: Vec<(Pairs<'_, Rule>, SyntaxId)> = vec![(root_pair.into_inner(), SyntaxId(0))];
501
502    while !frames.is_empty() {
503        let (next, parent) = {
504            let (pairs, parent) = frames.last_mut().unwrap();
505            (pairs.next(), *parent)
506        };
507        if let Some(pair) = next {
508            let pair_span = span(&pair);
509            match classify(&pair, &mut builder, preserve_comments) {
510                Action::Leaf(kind) => {
511                    builder.add(parent, kind, pair_span);
512                }
513                Action::Open(kind) => {
514                    let id = builder.add(parent, kind, pair_span);
515                    frames.push((pair.into_inner(), id));
516                }
517                Action::Transparent => {
518                    let inner = pair.into_inner();
519                    if inner.peek().is_some() {
520                        frames.push((inner, parent));
521                    }
522                }
523                Action::Ignore => {}
524            }
525        } else {
526            frames.pop();
527        }
528    }
529
530    Ok(builder)
531}

547fn build_parallel(source: &str, preserve_comments: bool, threads: usize) -> Result<Builder> {
548    let items = split_top_level(source)?;
549    let worker_count = threads.min(items.len()).max(1);
550    let pool = rayon::ThreadPoolBuilder::new()
551        .num_threads(worker_count)
552        .build()
553        .map_err(|error| Error::new(&format!("Unable to create ICL parser workers: {error}")))?;
554    let mut parsed_items: Vec<_> = pool.install(|| {
555        items
556            .par_iter()
557            .map(|item| {
558                let rule = match item.kind {
559                    TopLevelKind::Module => Rule::module_source,
560                    TopLevelKind::NameSpace => Rule::namespace_source,
561                    TopLevelKind::UseNameSpace => Rule::use_namespace_source,
562                };
563                let slice = &source[item.start..item.end];
564                let result = ICLParser::parse(rule, slice)
565                    .map_err(|error| Error::new(&error.to_string()))
566                    .and_then(|pairs| build(slice, pairs, preserve_comments));
567                (item.start, result)
568            })
569            .collect()
570    });
571    parsed_items.sort_by_key(|(offset, _)| *offset);
572
573    let mut builder = Builder::new(source.len());
574    for (offset, result) in parsed_items {
575        match result {
576            Ok(item) => builder.merge(item, offset as u32),
577            Err(error) => {
578                let line = source[..offset]
579                    .bytes()
580                    .filter(|byte| *byte == b'\n')
581                    .count()
582                    + 1;
583                return Err(Error::new(&format!(
584                    "Error in top-level ICL item beginning near line {line}:\n{}",
585                    error.msg
586                )));
587            }
588        }
589    }
590    Ok(builder)
591}

593fn split_top_level(source: &str) -> Result<Vec<TopLevelItem>> {
594    let bytes = source.as_bytes();
595    let mut items: Vec<TopLevelItem> = Vec::new();
596    let mut cursor = 0usize;
597    while cursor < bytes.len() {
598        let item_start = cursor;
599        let token_start = skip_layout(bytes, cursor)?;
600        if token_start == bytes.len() {
601            if let Some(last) = items.last_mut() {
602                last.end = bytes.len();
603            }
604            break;
605        }
606        let mut token_end = token_start;
607        while token_end < bytes.len()
608            && (bytes[token_end].is_ascii_alphanumeric() || bytes[token_end] == b'_')
609        {
610            token_end += 1;
611        }
612        let kind = match &source[token_start..token_end] {
613            "Module" => TopLevelKind::Module,
614            "NameSpace" => TopLevelKind::NameSpace,
615            "UseNameSpace" => TopLevelKind::UseNameSpace,
616            token => {
617                return Err(Error::new(&format!(
618                    "Unexpected top-level ICL token {token:?} at byte {token_start}"
619                )))
620            }
621        };
622        let end = scan_item_end(bytes, token_end, matches!(kind, TopLevelKind::Module))?;
623        items.push(TopLevelItem {
624            start: item_start,
625            end,
626            kind,
627        });
628        cursor = end;
629    }
630    if items.is_empty() {
631        return Err(Error::new(
632            "ICL source must contain at least one source item",
633        ));
634    }
635    Ok(items)
636}

638fn skip_layout(bytes: &[u8], mut cursor: usize) -> Result<usize> {
639    loop {
640        while cursor < bytes.len() && bytes[cursor].is_ascii_whitespace() {
641            cursor += 1;
642        }
643        if bytes.get(cursor..cursor + 2) == Some(b"//") {
644            cursor += 2;
645            while cursor < bytes.len() && bytes[cursor] != b'\n' {
646                cursor += 1;
647            }
648        } else if bytes.get(cursor..cursor + 2) == Some(b"/*") {
649            let Some(end) = bytes[cursor + 2..]
650                .windows(2)
651                .position(|window| window == b"*/")
652            else {
653                return Err(Error::new("Unterminated block comment in ICL source"));
654            };
655            cursor += end + 4;
656        } else {
657            return Ok(cursor);
658        }
659    }
660}

662fn scan_item_end(bytes: &[u8], mut cursor: usize, is_module: bool) -> Result<usize> {
663    let mut depth = 0usize;
664    let mut saw_open = false;
665    while cursor < bytes.len() {
666        match bytes[cursor] {
667            b'"' => {
668                cursor += 1;
669                while cursor < bytes.len() {
670                    match bytes[cursor] {
671                        b'\\' => cursor = (cursor + 2).min(bytes.len()),
672                        b'"' => {
673                            cursor += 1;
674                            break;
675                        }
676                        _ => cursor += 1,
677                    }
678                }
679            }
680            b'/' if bytes.get(cursor + 1) == Some(&b'/') => {
681                cursor += 2;
682                while cursor < bytes.len() && bytes[cursor] != b'\n' {
683                    cursor += 1;
684                }
685            }
686            b'/' if bytes.get(cursor + 1) == Some(&b'*') => {
687                let Some(end) = bytes[cursor + 2..]
688                    .windows(2)
689                    .position(|window| window == b"*/")
690                else {
691                    return Err(Error::new("Unterminated block comment in ICL source"));
692                };
693                cursor += end + 4;
694            }
695            b'{' if is_module => {
696                saw_open = true;
697                depth += 1;
698                cursor += 1;
699            }
700            b'}' if is_module => {
701                if depth == 0 {
702                    return Err(Error::new("Unexpected closing brace in ICL source"));
703                }
704                depth -= 1;
705                cursor += 1;
706                if saw_open && depth == 0 {
707                    return Ok(cursor);
708                }
709            }
710            b';' if !is_module => return Ok(cursor + 1),
711            _ => cursor += 1,
712        }
713    }
714    Err(Error::new("Unterminated top-level ICL source item"))
715}

730fn classify(pair: &Pair<'_, Rule>, builder: &mut Builder, preserve_comments: bool) -> Action {
731    let text = pair.as_str();
732    let kind = match pair.as_rule() {
733        Rule::pdl_statement | Rule::iproc_def | Rule::pdl_hash_comment => return Action::Ignore,
734        Rule::namespace_def => SyntaxKind::NameSpace,
735        Rule::use_namespace_def => SyntaxKind::UseNameSpace,
736        Rule::module_def => SyntaxKind::Module,
737        Rule::scan_in_port_def => SyntaxKind::Port(PortType::ScanIn),
738        Rule::scan_out_port_def => SyntaxKind::Port(PortType::ScanOut),
