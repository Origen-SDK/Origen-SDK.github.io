5/// Allows for arbitrary "data sources". This could be, for example, a database (any), LDAP, just a random file, etc.
6/// Just need a few methods implemented for frontend <-> backend integration. Others are available but are optional.
7pub trait DataStoreCategoryFrontendAPI {
8    fn name(&self) -> &str;
9    fn get_data_store(&self, store: &str) -> Result<Option<Box<dyn DataStoreFrontendAPI>>>;
10    fn add_data_store(
11        &self,
12        name: &str,
13        parameters: TypedValueMap,
14        backend_details: Option<TypedValueMap>,
15    ) -> Result<Box<dyn DataStoreFrontendAPI>>;
16
17    fn data_stores(&self) -> Result<HashMap<String, Box<dyn DataStoreFrontendAPI>>>;
18
19    fn require_data_store(&self, store: &str) -> Result<Box<dyn DataStoreFrontendAPI>> {
20        match self.get_data_store(store)? {
21            Some(ds) => Ok(ds),
22            None => bail!(
23                "Required data store '{}' not found in category '{}'",
24                store,
25                self.name()
26            ),
27        }
28    }
29
30    fn available_data_stores(&self) -> Result<Vec<String>> {
31        Ok(self
32            .data_stores()?
33            .keys()
34            .map(|k| k.to_string())
35            .collect::<Vec<String>>())
36    }
37
38    fn contains_data_store(&self, store: &str) -> Result<bool> {
39        Ok(self.data_stores()?.contains_key(store))
40    }
41
42    fn remove_data_store(&self, store: &str) -> Result<()>;
43}
