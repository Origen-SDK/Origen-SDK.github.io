24/// Render the current program as UltraFLEX IG-XL text worksheets.
25///
26/// Flow, Test Instances and Pattern Sets are emitted as independent importable
27/// sheets.  This deliberately uses the tester-neutral PGM AST rather than
28/// exposing a second UltraFLEX-specific flow API.
29pub fn render(output_dir: &Path) -> Result<(Vec<PathBuf>, Model)> {
30    std::fs::create_dir_all(output_dir)?;
31    let mut generated = vec![];
32    let mut resource_rows = vec![];
33    let mut referenced_patterns = vec![];
34    let model = FLOW.with_all_flows(|flows| {
35        let mut model = Model::new(SupportedTester::ULTRAFLEX);
36        for (name, flow) in flows {
37            let (ast, next_model) = process_flow(flow, model, SupportedTester::ULTRAFLEX, true)?;
38            if PROG_GEN_CONFIG.flow_visualization_enabled() {
39                generated.append(&mut render_flow_visualization(
40                    name,
41                    SupportedTester::ULTRAFLEX,
42                    &ast,
43                    &next_model,
44                    output_dir,
45                )?);
46            }
47            let (next_model, mut files, mut rows, mut patterns) =
48                render_flow(&ast, output_dir, next_model, name)?;
49            model = next_model;
50            generated.append(&mut files);
51            resource_rows.append(&mut rows);
52            referenced_patterns.append(&mut patterns);
53        }
54        Ok(model)
55    })?;
56    generated.append(&mut ResourceGenerator::new(resource_rows).render(output_dir)?);
57    if let Some(path) = write_referenced_list(output_dir, referenced_patterns)? {
58        generated.push(path);
59    }
60    Ok((generated, model))
61}

63fn render_flow(
64    ast: &Node<PGM>,
65    output_dir: &Path,
66    model: Model,
67    flow_name: &str,
68) -> Result<(Model, Vec<PathBuf>, Vec<ResourceRow>, Vec<String>)> {
69    let mut generator = FlowGenerator::new(model);
70    ast.process(&mut generator)?;
71
72    let flow_path = output_dir.join(format!("{}_flow.txt", flow_name));
73    write_sheet(&flow_path, &FLOW_HEADER, &generator.rows)?;
74
75    let instance_path = output_dir.join(format!("{}_instances.txt", flow_name));
76    generator.write_instances(&instance_path, flow_name)?;
77
78    let patset_path = output_dir.join(format!("{}_patsets.txt", flow_name));
79    generator.write_patsets(&patset_path)?;
80
81    let mut files = vec![flow_path, instance_path, patset_path];
82    let patgroups_path = output_dir.join(format!("{}_patgroups.txt", flow_name));
83    if generator.write_patgroups(&patgroups_path)? {
84        files.push(patgroups_path);
85    }
86    let patsubrs_path = output_dir.join(format!("{}_patsubrs.txt", flow_name));
87    if generator.write_patsubrs(&patsubrs_path)? {
88        files.push(patsubrs_path);
89    }
90
91    let patterns = generator
92        .patsets
93        .values()
94        .flat_map(|group| group.patterns.iter())
95        .map(|pattern| {
96            Path::new(&pattern.path)
97                .file_stem()
98                .and_then(|name| name.to_str())
99                .unwrap_or(&pattern.path)
100                .to_string()
101        })
102        .collect();
103    Ok((generator.model, files, generator.resources_rows, patterns))
104}

106fn write_sheet(path: &Path, header: &[&str], rows: &[String]) -> Result<()> {
107    let mut file = std::fs::File::create(path)?;
108    for line in header {
109        writeln!(file, "{}", line)?;
110    }
111    for row in rows {
112        writeln!(file, "{}", row)?;
113    }
114    Ok(())
115}
