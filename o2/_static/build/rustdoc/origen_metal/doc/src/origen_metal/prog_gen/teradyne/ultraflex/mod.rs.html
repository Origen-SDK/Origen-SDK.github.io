23/// Render the current program as UltraFLEX IG-XL text worksheets.
24///
25/// Flow, Test Instances and Pattern Sets are emitted as independent importable
26/// sheets.  This deliberately uses the tester-neutral PGM AST rather than
27/// exposing a second UltraFLEX-specific flow API.
28pub fn render(output_dir: &Path) -> Result<(Vec<PathBuf>, Model)> {
29    std::fs::create_dir_all(output_dir)?;
30    let mut generated = vec![];
31    let mut resource_rows = vec![];
32    let mut referenced_patterns = vec![];
33    let model = FLOW.with_all_flows(|flows| {
34        let mut model = Model::new(SupportedTester::ULTRAFLEX);
35        for (name, flow) in flows {
36            let (ast, next_model) = process_flow(flow, model, SupportedTester::ULTRAFLEX, true)?;
37            let (next_model, mut files, mut rows, mut patterns) =
38                render_flow(&ast, output_dir, next_model, name)?;
39            model = next_model;
40            generated.append(&mut files);
41            resource_rows.append(&mut rows);
42            referenced_patterns.append(&mut patterns);
43        }
44        Ok(model)
45    })?;
46    generated.append(&mut ResourceGenerator::new(resource_rows).render(output_dir)?);
47    if let Some(path) = write_referenced_list(output_dir, referenced_patterns)? {
48        generated.push(path);
49    }
50    Ok((generated, model))
51}

53fn render_flow(
54    ast: &Node<PGM>,
55    output_dir: &Path,
56    model: Model,
57    flow_name: &str,
58) -> Result<(Model, Vec<PathBuf>, Vec<ResourceRow>, Vec<String>)> {
59    let mut generator = FlowGenerator::new(model);
60    ast.process(&mut generator)?;
61
62    let flow_path = output_dir.join(format!("{}_flow.txt", flow_name));
63    write_sheet(&flow_path, &FLOW_HEADER, &generator.rows)?;
64
65    let instance_path = output_dir.join(format!("{}_instances.txt", flow_name));
66    generator.write_instances(&instance_path, flow_name)?;
67
68    let patset_path = output_dir.join(format!("{}_patsets.txt", flow_name));
69    generator.write_patsets(&patset_path)?;
70
71    let mut files = vec![flow_path, instance_path, patset_path];
72    let patgroups_path = output_dir.join(format!("{}_patgroups.txt", flow_name));
73    if generator.write_patgroups(&patgroups_path)? {
74        files.push(patgroups_path);
75    }
76    let patsubrs_path = output_dir.join(format!("{}_patsubrs.txt", flow_name));
77    if generator.write_patsubrs(&patsubrs_path)? {
78        files.push(patsubrs_path);
79    }
80
81    let patterns = generator
82        .patsets
83        .values()
84        .flat_map(|group| group.patterns.iter())
85        .map(|pattern| {
86            Path::new(&pattern.path)
87                .file_stem()
88                .and_then(|name| name.to_str())
89                .unwrap_or(&pattern.path)
90                .to_string()
91        })
92        .collect();
93    Ok((generator.model, files, generator.resources_rows, patterns))
94}

96fn write_sheet(path: &Path, header: &[&str], rows: &[String]) -> Result<()> {
97    let mut file = std::fs::File::create(path)?;
98    for line in header {
99        writeln!(file, "{}", line)?;
100    }
101    for row in rows {
102        writeln!(file, "{}", row)?;
103    }
104    Ok(())
105}
